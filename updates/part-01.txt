# Edusphere Overseas — website and portal

The new Edusphere Overseas site, built to the **Website Plan, Version 1**.

**Stack:** Next.js 16 (App Router, TypeScript) · Tailwind CSS 4 · Motion · Sanity (content) · Supabase (portal) · Cloudflare R2 (documents) · Vercel.
Coming in later steps: Supabase (portal, Mumbai region), Cloudflare R2 and Turnstile, Resend.

**Live preview:** on Vercel (project `edusphere-platform`); search engines are kept out until launch.
**Admin portal:** `/admin` → **Content** at `/admin/content` (Sanity Studio; sign in with your invited Sanity account)

## Run it on your computer

You need Node.js 22 or newer.

```bash
npm install
npm run dev        # http://localhost:3000
npm run build      # production build, the same one Vercel runs
```

## Where things live

| What | Where |
| --- | --- |
| Banners, ticker, offices, contact details, social links, destinations, services, "Open now", UK onshore programmes, team, office photos, section switches, hero text | **Admin → Content** (Sanity) |
| Built-in fallback content (used until Content is set up) | `src/config/site.ts` |
| Content types (what fields the team can edit) | `src/sanity/schemaTypes/` |
| Content menu layout | `src/sanity/structure.ts` |
| How the site reads content | `src/sanity/queries.ts`, `src/lib/content.ts` |
| Brand colours, animations | `src/app/globals.css` |
| Logo (vector) and loading animation | `src/components/brand/` |
| Home page sections | `src/components/home/` |
| Header, footer, ticker, WhatsApp button | `src/components/layout/` |
| Country flags (MIT licence) | `public/flags/` |

Published changes appear on the website within about a minute. For instant updates, add a Sanity webhook to `/api/revalidate` with a secret, and the same secret as `SANITY_REVALIDATE_SECRET` in Vercel.

To give a team member access to Content, invite them to the **Edusphere Platform** project at sanity.io/manage → Members.

## Logo loading animation

Plays once per browser session (about 1.6 seconds), then glides into the header logo. Tap to skip. It is skipped for visitors who have "reduce motion" turned on, and it hides itself even if JavaScript is slow.

## Deploy (Vercel)

1. Vercel is connected to this GitHub repository (project `edusphere-platform`). Every push to `main` deploys production; other branches get preview deployments.
2. Keys go in Vercel under **Project Settings → Environment Variables** (see `.env.example`), then redeploy.
3. Commits that only add files under `updates/` (update-bundle parts) are not built (`vercel.json` → `ignoreCommand`).

`netlify.toml` is kept only for the earlier Netlify project and can be removed once Netlify is switched off.

The site is set to `noindex` until launch; switch `robots` in `src/app/layout.tsx` at go-live (Step 8).

import { revalidateTag } from "next/cache";
import type { NextRequest } from "next/server";
import { parseBody } from "next-sanity/webhook";
import { SANITY_TAG } from "@/sanity/tags";

/**
 * Called by a Sanity webhook whenever content is published, so the website
 * updates straight away instead of within the usual minute.
 * Requires SANITY_REVALIDATE_SECRET in Vercel and the same secret on the webhook.
 */
export async function POST(req: NextRequest) {
  const secret = process.env.SANITY_REVALIDATE_SECRET;
  if (!secret) return new Response("Revalidation is not configured", { status: 501 });

  const { isValidSignature, body } = await parseBody<{ _type?: string }>(req, secret);
  if (!isValidSignature) return new Response("Invalid signature", { status: 401 });

  revalidateTag(SANITY_TAG, { expire: 0 });
  return Response.json({ revalidated: true, type: body?._type ?? null });
}

import type { MetadataRoute } from "next";

/**
 * Search engines are kept out until launch (Step 7). At launch, set
 * SITE_LAUNCHED=true in Vercel (Project Settings → Environment Variables) and remove `robots` noindex in app/layout.tsx.

export default function robots(): MetadataRoute.Robots {
  if (process.env.SITE_LAUNCHED !== "true") return { rules: { userAgent: "*", disallow: "/" } };
  return {
    rules: { userAgent: "*", allow: "/", disallow: ["/admin", "/portal", "/api", "/sign-in", "/sign-up", "/forgot-password", "/reset-password", "/welcome", "/auth"] },
    sitemap: `${site.url}/sitemap.xml`,
    host: site.url,
  };
}

/** The hero "sphere": the Edusphere mark at the centre of orbiting flags. */
export function FlagGlobe() {
  const mx = useMotionValue(0);
  const my = useMotionValue(0);
  const sx = useSpring(mx, { stiffness: 60, damping: 18 });
  const sy = useSpring(my, { stiffness: 60, damping: 18 });
  const rotateY = useTransform(sx, [-0.5, 0.5], [10, -10]);
  const rotateX = useTransform(sy, [-0.5, 0.5], [-8, 8]);
  const glowX = useTransform(sx, [-0.5, 0.5], ["40%", "60%"]);
  const glowY = useTransform(sy, [-0.5, 0.5], ["35%", "55%"]);

  return (
    <div
      className="globe relative mx-auto aspect-square w-[86%] max-w-[34rem] sm:w-full [perspective:1200px]"
      onPointerMove={(e) => {
        const r = e.currentTarget.getBoundingClientRect();
        mx.set((e.clientX - r.left) / r.width - 0.5);
        my.set((e.clientY - r.top) / r.height - 0.5);
      }}
      onPointerLeave={() => {
        mx.set(0);
        my.set(0);
      }}
    >
      <motion.div className="absolute inset-0" style={{ rotateX, rotateY, transformStyle: "preserve-3d" }}>
        {/* glow */}
        <div className="glow-pulse absolute inset-[18%] rounded-full bg-[radial-gradient(circle,rgba(92,192,207,0.35),rgba(90,155,209,0.12)_45%,transparent_70%)] blur-2xl" />

export type EnquiryResult = { ok: true } | { ok: false; reason: "invalid" | "unavailable" | "busy" };

const text = (form: FormData, key: string, max: number) =>
  String(form.get(key) ?? "")
    .replace(/\s+/g, " ")
    .trim()
    .slice(0, max);

export async function submitEnquiry(form: FormData): Promise<EnquiryResult> {
  // Bots fill the hidden field; pretend it worked and keep nothing.
  if (text(form, "bot-field", 200)) return { ok: true };

  const name = text(form, "name", 120);
  const phone = text(form, "phone", 30);
  const email = text(form, "email", 254).toLowerCase();
  const message = String(form.get("message") ?? "").trim().slice(0, 3500);
  const page = text(form, "page", 200);
  const details = ["destination", "level", "intake", "office"].map((k) => text(form, k, 80)).filter(Boolean);

  if (
    !name ||
    !/^[0-9+()\-\s]{7,30}$/.test(phone) ||
    (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) ||
    form.get("consent") !== "yes"
  ) {
    return { ok: false, reason: "invalid" };
  }

  if (!portalConfigured || !(process.env.SUPABASE_SECRET_KEY || process.env.SUPABASE_SERVICE_ROLE_KEY)) {
    return { ok: false, reason: "unavailable" };
  }

  try {
    const db = createAdminClient();

    // Flood guard: at most 3 enquiries from the same phone number in 10 minutes.
    const since = new Date(Date.now() - 10 * 60 * 1000).toISOString();
    const { count } = await db
      .from("leads")
      .select("id", { count: "exact", head: true })
      .eq("source", "website_enquiry")
      .eq("phone", phone)
      .gte("created_at", since);
    if ((count ?? 0) >= 3) return { ok: false, reason: "busy" };

    const interest = details.join(" · ").slice(0, 200) || null;
    const { error } = await db.from("leads").insert({
      source: "website_enquiry",
      name,
      phone,
      email: email || null,
      interest,
      message: [message, page && `Sent from ${page}`].filter(Boolean).join("\n\n") || null,
    });
    if (error) {
      console.error("[enquiry] could not save:", error.message);
      return { ok: false, reason: "unavailable" };
    }

    await sendEmail({
      to: staffInbox(),
      replyTo: email || undefined,
      subject: `New website enquiry: ${name}`,
      heading: "New website enquiry",
      lines: [
        `${name} · ${phone}${email ? ` · ${email}` : ""}`,
        interest ?? "",
        message ? (message.length > 600 ? message.slice(0, 597) + "…" : message) : "",
        "It is saved in the enquiry inbox (Supabase → Table Editor → leads).",
      ].filter(Boolean),
    });
    return { ok: true };
  } catch (error) {
    console.error("[enquiry] failed:", error instanceof Error ? error.message : error);
    return { ok: false, reason: "unavailable" };
  }
}
