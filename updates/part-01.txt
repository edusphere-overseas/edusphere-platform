async function Apply({ searchParams }: { searchParams: PageProps<"/apply">["searchParams"] }) {
  const sp = await searchParams;
  await connection(); // the intake list depends on today's date
  const one = (k: string) => (typeof sp[k] === "string" ? (sp[k] as string).slice(0, 200) : "");
  const u = one("u") ? await getUniversity(one("u")) : null;
  const country = u?.country ?? (isCountry(one("country")) ? one("country") : "");
  const initial = {
    country,
    university: u ? { id: u.id, name: u.name, place: [u.city, u.region].filter(Boolean).join(", ") } : null,
    typed: u ? "" : one("name").slice(0, 160),
    course: one("course"),
  };
  const intakes = intakeOptions();

  // Signed in: add straight to the portal's application tracker
  if (portalConfigured) {
    const supabase = await createClient();
    const { data } = await supabase.auth.getClaims();
    const uid = data?.claims?.sub;
    if (uid) {
      const { data: profile } = await supabase.from("profiles").select("role, full_name, onboarded_at").eq("id", uid).maybeSingle();
      if (profile?.role === "admin" || profile?.role === "counsellor" || profile?.role === "editor") {
        return (
          <div>
            <h2 className="text-xl font-extrabold text-deep">You&apos;re signed in as staff</h2>
            <p className="mt-2 leading-relaxed text-slate">
              Add universities for a student from their page in the{" "}
              <Link href="/staff/students" className="font-semibold text-teal hover:text-navy">
                staff dashboard
              </Link>
              . Sign out to try the student form.
            </p>
          </div>
        );
      }
      if (profile?.onboarded_at) {
        return (
          <>
            <h2 className="flex items-center gap-2 text-xl font-extrabold text-deep">
              <CheckCircle2 className="h-5 w-5 text-teal" aria-hidden="true" />
              Signed in as {profile.full_name || "you"}
            </h2>
            <p className="mt-2 mb-6 leading-relaxed text-slate">
              {profile.role === "parent"
                ? "The university is added to your student's application tracker, and their counsellor is told."
                : "The university is added to your application tracker, and your counsellor is told."}
            </p>
            <AddUniversityForm countries={countries} intakes={intakes} initial={initial} />
          </>
        );
      }
    }
  }

  return <ApplyForm countries={countries} intakes={intakes} initial={initial} signInHref={`/sign-in?next=${encodeURIComponent(`/apply${u ? `?u=${u.id}` : ""}`)}`} />;
}

import { btnPrimary, hint, input, label } from "@/components/portal/ui";
import { consentText } from "@/lib/portal/consent";
import { UniversityFields, type CountryOption, type PickedUniversity } from "./UniversityFields";

/** The course finder's "Apply" form for visitors: their choice, their details, and a new portal account. */
export function ApplyForm({
  countries,
  intakes,
  initial,
  signInHref = "/sign-in",
}: {
  countries: CountryOption[];
  intakes: string[];
  initial: { country?: string; university?: PickedUniversity; typed?: string; course?: string };
  signInHref?: string;
}) {
  const [state, action, pending] = useActionState<ApplyState, FormData>(applyForUniversity, null);
  const v = state?.values;

  if (state?.done) {
    return (
      <div className="text-center" role="status">
        <div className="mx-auto grid h-14 w-14 place-items-center rounded-2xl bg-teal/12 text-teal">
          <MailCheck className="h-7 w-7" aria-hidden="true" />
        </div>
        <h2 className="mt-4 text-xl font-extrabold text-deep">Application received</h2>
        <p className="mx-auto mt-3 max-w-md leading-relaxed text-slate">
          Thanks! Your counsellor will be in touch about <strong className="text-deep">{state.done.university}</strong>, usually within one working day.
        </p>
        <p className="mx-auto mt-3 max-w-md leading-relaxed text-slate">
          We&apos;ve also sent a link to <strong className="text-deep">{state.done.email}</strong>. Open it to verify your email and choose a password for your
          student portal, where you can follow every stage of your application.
        </p>
        <p className="mt-3 text-sm text-slate">Nothing after a few minutes? Check your spam folder.</p>
        <p className="mt-6 text-sm text-slate">
          Already have an account?{" "}
          <Link href="/sign-in?next=/portal/applications" className="font-semibold text-teal hover:text-navy">
            Sign in to see your applications
          </Link>
        </p>
      </div>
    );
  }

  return (
    <form
      noValidate={false}
      onSubmit={(e) => {
        e.preventDefault();
        const data = new FormData(e.currentTarget);
        startTransition(() => action(data));
      }}
      className="grid gap-8"
    >
      <fieldset className="grid gap-4">
        <legend className="mb-4 flex items-center gap-3 text-lg font-extrabold text-deep">
          <span className="grid h-7 w-7 place-items-center rounded-full bg-teal text-sm text-white">1</span>

        </legend>
        <UniversityFields
          countries={countries}
          intakes={intakes}
          initialCountry={v?.country ?? initial.country ?? ""}
          initialUniversity={initial.university ?? null}
          initialTyped={v?.university ?? initial.typed ?? ""}

          initialIntake={v?.intake ?? ""}

        />
      </fieldset>

        </legend>

          Full name
          <input name="full_name" required autoComplete="name" maxLength={120} defaultValue={v?.full_name} aria-invalid={state?.field === "full_name" || undefined} className={input} />
        </label>
        <div className="grid gap-4 sm:grid-cols-2">
          <label className={label}>
            Date of birth
            <input
              name="date_of_birth"
              type="date"
              required
              max={new Date().toISOString().slice(0, 10)}
              defaultValue={v?.date_of_birth}
              aria-invalid={state?.field === "date_of_birth" || undefined}
              className={input}
            />
          </label>
          <label className={label}>
            Mobile number
            <input
              name="phone"
              type="tel"
              required
              autoComplete="tel"
              inputMode="tel"
              maxLength={30}
              placeholder="+91 …"
              defaultValue={v?.phone}
              aria-invalid={state?.field === "phone" || undefined}
              className={input}
            />
          </label>
        </div>
        <label className={label}>
          Email
          <input name="email" type="email" required autoComplete="email" maxLength={254} defaultValue={v?.email} aria-invalid={state?.field === "email" || undefined} className={input} />
          <span className={hint}>We create your student portal account with this email, so you can follow your application.</span>

        <Link href={signInHref} className="font-semibold text-teal hover:text-navy">
          Sign in
        </Link>{" "}
        and add universities from your portal.
      </p>
    </form>
  );
}
