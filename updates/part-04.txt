export function checklistWithStatus(items: ChecklistItem[], docs: DocRow[]) {
  const general = docs.filter((d) => !d.service_request_id);
  const entries: ChecklistEntry[] = items.map((item) => {
    const files = general.filter((d) => d.doc_type === item.key);
    return { ...item, files, status: itemStatus(files) };
  });
  // Documents of a kind that's no longer on the checklist still show, under "Other documents"
  const known = new Set(items.map((i) => i.key));
  const others = general.filter((d) => !known.has(d.doc_type));
  const otherEntry: ChecklistEntry = { ...OTHER_DOC, files: others, status: itemStatus(others) };
  const required = entries.filter((e) => e.required);
  return {
    entries,
    other: otherEntry,
    requiredCount: required.length,
    approvedCount: required.filter((e) => e.status === "approved").length,
    reuploadCount: entries.filter((e) => e.status === "reupload").length + (otherEntry.status === "reupload" ? 1 : 0),
    reviewCount: entries.filter((e) => e.status === "under_review").length + (otherEntry.status === "under_review" ? 1 : 0),
  };
}

export async function sendEmail(email: Email) {
  const to = (Array.isArray(email.to) ? email.to : [email.to]).filter(Boolean);
  if (!to.length) return;
  const { html, text } = renderEmail(email);
  const key = process.env.RESEND_API_KEY;
  const from = process.env.EMAIL_FROM;

  if (!key || !from) {
    console.info(`[email] (not sent: no RESEND_API_KEY/EMAIL_FROM) to=${to.join(",")} subject="${email.subject}"`);
    if (process.env.EMAIL_OUTBOX) {
      await appendFile(process.env.EMAIL_OUTBOX, JSON.stringify({ to, subject: email.subject, text }) + "\n").catch(() => {});
    }
    return;
  }
  try {
    const { error } = await new Resend(key).emails.send({ from, to, subject: email.subject, html, text, replyTo: email.replyTo });
    if (error) console.error("[email] failed:", error.message);
  } catch (error) {
    console.error("[email] failed:", error instanceof Error ? error.message : error);
  }
}

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";
import { portalConfigured, supabaseKey, supabaseUrl } from "@/lib/supabase/env";

/**
 * Keeps portal sign-ins fresh (Supabase sessions are short-lived and renewed here,
 * where cookies can be written) and sends signed-out visitors to the sign-in page.
 * Every page and action still checks access itself; this is not the only guard.
 */
export async function proxy(request: NextRequest) {
  if (!portalConfigured) return NextResponse.next({ request });

  let response = NextResponse.next({ request });
  const supabase = createServerClient(supabaseUrl, supabaseKey, {
    cookies: {
      getAll: () => request.cookies.getAll(),
      setAll(cookiesToSet, headers) {
        cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value));
        response = NextResponse.next({ request });
        cookiesToSet.forEach(({ name, value, options }) => response.cookies.set(name, value, options));
        Object.entries(headers ?? {}).forEach(([key, value]) => response.headers.set(key, value));
      },
    },
  });

  const { data } = await supabase.auth.getClaims();
  const signedIn = Boolean(data?.claims?.sub);
  const { pathname, search } = request.nextUrl;

  const isPrivate = ["/portal", "/staff"].some((p) => pathname === p || pathname.startsWith(p + "/")) || pathname === "/welcome";
  if (!signedIn && isPrivate) {
    const url = request.nextUrl.clone();
    url.pathname = "/sign-in";
    url.search = "";
    url.searchParams.set("next", pathname + search);
    return redirectWith(response, url);
  }
  if (signedIn && (pathname === "/sign-in" || pathname === "/sign-up")) {
    const url = request.nextUrl.clone();
    url.pathname = "/portal";
    url.search = "";
    return redirectWith(response, url);
  }
  return response;
}

/** Redirect while keeping any refreshed session cookies. */
function redirectWith(response: NextResponse, url: URL) {
  const redirect = NextResponse.redirect(url);
  response.cookies.getAll().forEach((c) => redirect.cookies.set(c));
  redirect.headers.set("Cache-Control", "private, no-store");
  return redirect;
