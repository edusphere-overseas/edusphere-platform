import type { Metadata } from "next";
import { redirect } from "next/navigation";
import { Suspense } from "react";
import { WelcomeForm } from "@/components/portal/AuthForms";
import { PortalUnavailable } from "@/components/portal/PortalUnavailable";
import { portalConfigured } from "@/lib/supabase/env";
import { createClient } from "@/lib/supabase/server";

export const metadata: Metadata = { title: "Welcome" };

/** First visit after Google sign-in or an invitation: choose student or parent, and consent. */
export default function WelcomePage({ searchParams }: PageProps<"/welcome">) {
  if (!portalConfigured) return <PortalUnavailable />;
  return (
    <Suspense fallback={<div className="h-[30rem] rounded-3xl bg-white/90" />}>
      <Welcome searchParams={searchParams} />
    </Suspense>
  );
}

async function Welcome({ searchParams }: { searchParams: PageProps<"/welcome">["searchParams"] }) {
  const supabase = await createClient();
  const { data } = await supabase.auth.getClaims();
  const uid = data?.claims?.sub;
  if (!uid) redirect("/sign-in");
  const [{ data: profile }, { data: details }] = await Promise.all([
    supabase.from("profiles").select("role, full_name, phone, onboarded_at, invited_by").eq("id", uid).single(),
    supabase.from("student_profiles").select("date_of_birth").eq("user_id", uid).maybeSingle(),
  ]);
  if (!profile) redirect("/sign-in");
  if (profile.onboarded_at || ["admin", "counsellor", "editor"].includes(profile.role ?? "")) redirect("/portal");
  const p = await searchParams;
  return (
    <WelcomeForm
      role={profile.role}
      suggestedRole={typeof p.role === "string" ? p.role : undefined}
      name={profile.full_name}
      phone={profile.phone ?? ""}
      dob={details?.date_of_birth ?? ""}
      invited={!!profile.invited_by}
    />
  );
}

import { NextResponse, type NextRequest } from "next/server";
import { safeNext } from "@/lib/portal/urls";
import { portalConfigured } from "@/lib/supabase/env";
import { createClient } from "@/lib/supabase/server";

/** Google sign-in returns here with a one-time code, which becomes a session. */
export async function GET(request: NextRequest) {
  const { searchParams, origin } = request.nextUrl;
  const code = searchParams.get("code");
  const next = safeNext(searchParams.get("next"));
  if (code && portalConfigured) {
    const supabase = await createClient();
    const { error } = await supabase.auth.exchangeCodeForSession(code);
    if (!error) return NextResponse.redirect(new URL(next, origin));
  }
  return NextResponse.redirect(new URL("/sign-in?error=google", origin));
}

import type { EmailOtpType } from "@supabase/supabase-js";
import { NextResponse, type NextRequest } from "next/server";
import { portalConfigured } from "@/lib/supabase/env";
import { createClient } from "@/lib/supabase/server";

const types: EmailOtpType[] = ["signup", "email", "recovery", "invite", "email_change", "magiclink"];

/**
 * Links in Supabase emails (verify email, reset password, accept an invitation) come here.
 * They carry a one-time token, so they work on any device, not just the one used to sign up.
 */
export async function GET(request: NextRequest) {
  const { searchParams, origin } = request.nextUrl;
  const tokenHash = searchParams.get("token_hash");
  const type = searchParams.get("type") as EmailOtpType | null;

  if (portalConfigured && tokenHash && type && types.includes(type)) {
    const supabase = await createClient();
    const { error } = await supabase.auth.verifyOtp({ type, token_hash: tokenHash });
    if (!error) {
      const to =
        type === "recovery" ? "/reset-password" : type === "invite" ? "/reset-password?welcome=1" : type === "email_change" ? "/portal/profile" : "/portal?verified=1";
      return NextResponse.redirect(new URL(to, origin));
    }
  }
  return NextResponse.redirect(new URL("/sign-in?error=link", origin));
}

import { NextResponse, type NextRequest } from "next/server";
import { logActivity } from "@/lib/portal/actor";
import { signedDownloadUrl, storageConfigured } from "@/lib/portal/storage";
import { portalConfigured } from "@/lib/supabase/env";
import { createClient } from "@/lib/supabase/server";

/**
 * Open a document: checks access (database rules), records the view in the activity log,
 * then sends the browser to a private link that expires after 5 minutes.
 */
export async function GET(request: NextRequest, { params }: RouteContext<"/portal/files/[id]">) {
  const { id } = await params;
  if (!portalConfigured) return NextResponse.redirect(new URL("/portal", request.url));
  const supabase = await createClient();
  const { data: claims } = await supabase.auth.getClaims();
  const uid = claims?.claims?.sub;
  if (!uid) return NextResponse.redirect(new URL("/sign-in", request.url));
  if (!/^[0-9a-f-]{36}$/i.test(id) || !storageConfigured()) return new NextResponse("Not found", { status: 404 });

  const { data: doc } = await supabase.from("documents").select("id, student_id, file_key, file_name, mime_type, status").eq("id", id).maybeSingle();
  if (!doc || doc.status === "uploading") return new NextResponse("Not found", { status: 404 });

  const download = request.nextUrl.searchParams.get("download") === "1";
  await logActivity(uid, doc.student_id, download ? "document_downloaded" : "document_viewed", "document", doc.id, { file: doc.file_name });
  const url = await signedDownloadUrl(doc.file_key, doc.file_name, doc.mime_type, download);
  const res = NextResponse.redirect(url);
  res.headers.set("Cache-Control", "private, no-store");
  res.headers.set("Referrer-Policy", "no-referrer");
  return res;
}
