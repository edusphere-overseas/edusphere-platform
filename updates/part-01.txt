"use server";

import { headers } from "next/headers";
import { redirect } from "next/navigation";
import { CONSENT_VERSION } from "@/lib/portal/consent";
import { safeNext } from "@/lib/portal/urls";
import { siteOrigin } from "@/lib/supabase/env";
import { createClient } from "@/lib/supabase/server";

export type FormState = {
  error?: string;
  done?: string;
  /** typed values sent back so the form keeps them after an error (never passwords) */
  values?: Record<string, string>;
  unverifiedEmail?: string;
} | null;

const str = (f: FormData, k: string) => String(f.get(k) ?? "").trim();
const emailOk = (e: string) => /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(e) && e.length <= 254;
const captcha = (f: FormData) => str(f, "cf-turnstile-response") || undefined;

function friendly(message: string) {
  if (/captcha/i.test(message)) return "The spam check didn't finish. Please wait a moment and try again.";
  if (/rate limit|too many/i.test(message)) return "Too many attempts. Please wait a few minutes and try again.";
  if (/weak|pwned|password should/i.test(message)) return "Please choose a stronger password (at least 8 characters, mixing letters and numbers).";
  return "Something went wrong. Please try again.";
}

export async function signIn(_: FormState, form: FormData): Promise<FormState> {
  const email = str(form, "email").toLowerCase();
  const password = String(form.get("password") ?? "");
  const values = { email };
  if (!emailOk(email) || !password) return { error: "Enter your email and password.", values };

  const supabase = await createClient();
  const { error } = await supabase.auth.signInWithPassword({ email, password, options: { captchaToken: captcha(form) } });
  if (error) {
    if (error.code === "email_not_confirmed") {
      return { error: "Please verify your email first: open the link we sent you.", values, unverifiedEmail: email };
    }
    if (error.code === "invalid_credentials") return { error: "That email and password don't match.", values };
    return { error: friendly(error.message), values };
  }
  redirect(safeNext(str(form, "next")));
}

export async function signUp(_: FormState, form: FormData): Promise<FormState> {
  const role = str(form, "role") === "parent" ? "parent" : "student";
  const values = {
    role,
    full_name: str(form, "full_name").slice(0, 120),
    email: str(form, "email").toLowerCase(),
    phone: str(form, "phone").slice(0, 30),
    date_of_birth: str(form, "date_of_birth"),
  };
  const password = String(form.get("password") ?? "");

  if (!values.full_name) return { error: "Please enter your full name.", values };
  if (!emailOk(values.email)) return { error: "Please enter a valid email address.", values };
  if (role === "student") {
    const dob = new Date(values.date_of_birth + "T00:00:00Z");
    if (!/^\d{4}-\d{2}-\d{2}$/.test(values.date_of_birth) || isNaN(+dob) || dob > new Date() || dob < new Date("1940-01-02")) {
      return { error: "Please enter your date of birth.", values };
    }
  }
  if (password.length < 8 || !/[A-Za-z]/.test(password) || !/\d/.test(password)) {
    return { error: "Use at least 8 characters for your password, with letters and numbers.", values };
  }
  if (form.get("consent") !== "on") return { error: "Please tick the consent box to create your account.", values };

  const supabase = await createClient();
  const { error } = await supabase.auth.signUp({
    email: values.email,
    password,
    options: {
      emailRedirectTo: `${siteOrigin()}/auth/confirm`,
      captchaToken: captcha(form),
      data: {
        role,
        full_name: values.full_name,
        phone: values.phone,
        date_of_birth: role === "student" ? values.date_of_birth : undefined,
        consent_version: CONSENT_VERSION,
      },
    },
  });
  if (error) {
    if (error.code === "user_already_exists") return { done: values.email };
    return { error: friendly(error.message), values };
  }
  // Same answer whether or not the email already had an account
  return { done: values.email };
}

export async function resendVerification(_: FormState, form: FormData): Promise<FormState> {
  const email = str(form, "email").toLowerCase();
  if (!emailOk(email)) return { error: "Please enter a valid email address." };
  const supabase = await createClient();
  const { error } = await supabase.auth.resend({
    type: "signup",
    email,
    options: { emailRedirectTo: `${siteOrigin()}/auth/confirm`, captchaToken: captcha(form) },
  });
  if (error && /rate|too many/i.test(error.message)) return { error: friendly(error.message) };
  return { done: email };
}

export async function signInWithGoogle(form: FormData) {
  const supabase = await createClient();
  const h = await headers();
  const host = h.get("x-forwarded-host") ?? h.get("host");
  const proto = h.get("x-forwarded-proto") ?? (host?.startsWith("localhost") || host?.startsWith("127.") ? "http" : "https");
  // Supabase only returns to addresses on its allow list, so the request's own address is safe here.
  const origin = host ? `${proto}://${host}` : siteOrigin();
  const next = safeNext(str(form, "next"));
  const role = str(form, "role");
  const target = new URL("/auth/callback", origin);
  target.searchParams.set("next", role === "student" || role === "parent" ? `/welcome?role=${role}` : next);

  const { data, error } = await supabase.auth.signInWithOAuth({
    provider: "google",
    options: { redirectTo: target.toString(), queryParams: { prompt: "select_account" } },
  });
  if (error || !data.url) redirect("/sign-in?error=google");
  redirect(data.url);
}

export async function requestPasswordReset(_: FormState, form: FormData): Promise<FormState> {
  const email = str(form, "email").toLowerCase();
  if (!emailOk(email)) return { error: "Please enter a valid email address.", values: { email } };
  const supabase = await createClient();
  const { error } = await supabase.auth.resetPasswordForEmail(email, {
    redirectTo: `${siteOrigin()}/auth/confirm`,
    captchaToken: captcha(form),
  });
  if (error && /captcha|rate|too many/i.test(error.message)) return { error: friendly(error.message), values: { email } };
  return { done: email };
}

export async function setNewPassword(_: FormState, form: FormData): Promise<FormState> {
  const password = String(form.get("password") ?? "");
  if (password.length < 8 || !/[A-Za-z]/.test(password) || !/\d/.test(password)) {
    return { error: "Use at least 8 characters, with letters and numbers." };
  }
  if (password !== String(form.get("confirm") ?? "")) return { error: "The two passwords don't match." };
  const supabase = await createClient();
  const { data } = await supabase.auth.getClaims();
  if (!data?.claims) return { error: "This link has expired. Please ask for a new one." };
  const { error } = await supabase.auth.updateUser({ password });
  if (error) {
    if (error.code === "same_password") return { error: "Please choose a password you haven't used here before." };
    return { error: friendly(error.message) };
  }
  redirect("/portal");
}

export async function completeOnboarding(_: FormState, form: FormData): Promise<FormState> {
  const role = str(form, "role");
  const values = {
    role,
    full_name: str(form, "full_name").slice(0, 120),
    phone: str(form, "phone").slice(0, 30),
    date_of_birth: str(form, "date_of_birth"),
  };
  if (role !== "student" && role !== "parent") return { error: "Please choose student or parent.", values };
  if (!values.full_name) return { error: "Please enter your full name.", values };
  if (role === "student" && !/^\d{4}-\d{2}-\d{2}$/.test(values.date_of_birth)) return { error: "Please enter your date of birth.", values };
  if (form.get("consent") !== "on") return { error: "Please tick the consent box to continue.", values };

  const supabase = await createClient();
  const { error } = await supabase.rpc("complete_onboarding", {
    p_role: role,
    p_full_name: values.full_name,
    p_phone: values.phone,
    p_date_of_birth: (role === "student" ? values.date_of_birth : null) as string,
    p_consent_version: CONSENT_VERSION,
  });
  if (error) return { error: error.code === "23514" || error.code === "42501" ? error.message : "Something went wrong. Please try again.", values };
  redirect("/portal");
}

export async function signOut() {
  const supabase = await createClient();
  await supabase.auth.signOut();
  redirect("/");
}

import { NextResponse, type NextRequest } from "next/server";
import { safeNext } from "@/lib/portal/urls";
import { createClient } from "@/lib/supabase/server";

/** Google sign-in returns here with a one-time code, which becomes a session. */
export async function GET(request: NextRequest) {
  const { searchParams, origin } = request.nextUrl;
  const code = searchParams.get("code");
  const next = safeNext(searchParams.get("next"));
  if (code) {
    const supabase = await createClient();
    const { error } = await supabase.auth.exchangeCodeForSession(code);
    if (!error) return NextResponse.redirect(new URL(next, origin));
  }
  return NextResponse.redirect(new URL("/sign-in?error=google", origin));
}

import type { EmailOtpType } from "@supabase/supabase-js";
import { NextResponse, type NextRequest } from "next/server";
import { createClient } from "@/lib/supabase/server";

const types: EmailOtpType[] = ["signup", "email", "recovery", "invite", "email_change", "magiclink"];

/**
 * Links in Supabase emails (verify email, reset password, accept an invitation) come here.
 * They carry a one-time token, so they work on any device, not just the one used to sign up.
 */
export async function GET(request: NextRequest) {
  const { searchParams, origin } = request.nextUrl;
  const tokenHash = searchParams.get("token_hash");
  const type = searchParams.get("type") as EmailOtpType | null;

  if (tokenHash && type && types.includes(type)) {
    const supabase = await createClient();
    const { error } = await supabase.auth.verifyOtp({ type, token_hash: tokenHash });
    if (!error) {
      const to =
        type === "recovery" ? "/reset-password" : type === "invite" ? "/reset-password?welcome=1" : type === "email_change" ? "/portal/profile" : "/portal?verified=1";
      return NextResponse.redirect(new URL(to, origin));
    }
  }
  return NextResponse.redirect(new URL("/sign-in?error=link", origin));
}

"use server";

import { refresh } from "next/cache";
import { getActor } from "@/lib/portal/actor";
import { ALLOWED_TYPES, MAX_FILE_BYTES } from "@/lib/portal/journey";
import { deleteObject, objectKey, safeFileName, signedUploadUrl, storageConfigured, verifyUpload } from "@/lib/portal/storage";
import { createAdminClient } from "@/lib/supabase/admin";

type StartInput = {
  studentId: string;
  docType: string;
  label?: string;
  fileName: string;
  mime: string;
  size: number;
  serviceRequestId?: string | null;
};

const uuid = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

/**
 * Step 1 of an upload: record it as "uploading" (the database checks this person may
 * upload for this student) and return a signed link the browser sends the file to.
 */
export async function startUpload(input: StartInput): Promise<{ id: string; url: string } | { error: string }> {
  const actor = await getActor();
  if (!actor) return { error: "Please sign in again." };
  if (!storageConfigured()) return { error: "Document uploads aren't switched on yet. Please try again soon." };
  if (!uuid.test(input.studentId) || !/^[a-z0-9_]{2,40}$/.test(input.docType)) return { error: "Something went wrong. Please refresh the page." };
  if (!(ALLOWED_TYPES as readonly string[]).includes(input.mime)) return { error: "Please upload a PDF, JPG or PNG file." };
  if (!Number.isInteger(input.size) || input.size < 1 || input.size > MAX_FILE_BYTES) return { error: "Files can be up to 10 MB." };
  if (input.serviceRequestId && !uuid.test(input.serviceRequestId)) return { error: "Something went wrong. Please refresh the page." };

  const id = crypto.randomUUID();
  const fileName = safeFileName(input.fileName, input.mime);
  const key = objectKey(input.studentId, id, fileName);
  const { error } = await actor.supabase.from("documents").insert({
    id,
    student_id: input.studentId,
    doc_type: input.docType,
    label: input.docType === "other" || input.docType === "service" ? (input.label ?? "").trim().slice(0, 120) || "Document" : null,
    file_key: key,
    file_name: fileName,
    mime_type: input.mime,
    size_bytes: input.size,
    uploaded_by: actor.uid,
    service_request_id: input.serviceRequestId || null,
  });
  if (error) {
    if (error.code === "42501") return { error: "You can't upload for this student yet. Students under 18 need an approved parent first." };
    return { error: "Something went wrong. Please try again." };
  }
  return { id, url: await signedUploadUrl(key, input.mime, input.size) };
}

/**
 * Step 2: the browser says the file arrived. Check it really is the signed size and a real
 * PDF/JPG/PNG, then send it for review. Files that fail the check are deleted.
 */
export async function finishUpload(id: string, replaces?: string | null): Promise<{ ok: true } | { error: string }> {
  const actor = await getActor();
  if (!actor || !uuid.test(id)) return { error: "Please sign in again." };
  const { data: doc } = await actor.supabase.from("documents").select("*").eq("id", id).maybeSingle();
  if (!doc || doc.status !== "uploading" || doc.uploaded_by !== actor.uid) return { error: "Upload not found. Please try again." };

  const admin = createAdminClient();
  const check = await verifyUpload(doc.file_key, doc.mime_type, doc.size_bytes);
  if (!check.ok) {
    await deleteObject(doc.file_key).catch(() => {});
    await admin.from("documents").delete().eq("id", id);
    return { error: check.reason };
  }

  // Replacing an earlier file of the same kind for the same student
  let replaced: string | null = null;
  if (replaces && uuid.test(replaces)) {
    const { data: old } = await actor.supabase.from("documents").select("id, student_id, doc_type, status").eq("id", replaces).maybeSingle();
    if (old && old.student_id === doc.student_id && old.doc_type === doc.doc_type && !["uploading", "replaced"].includes(old.status)) {
      replaced = old.id;
    }
  }

  // Staff uploads (offer letters and the like) are approved straight away
  const { error } = await admin
    .from("documents")
    .update(
      actor.staff
        ? { status: "approved", reviewed_by: actor.uid, reviewed_at: new Date().toISOString(), replaces: replaced }
        : { status: "under_review", replaces: replaced },
    )
    .eq("id", id)
    .eq("status", "uploading");
  if (error) return { error: "Something went wrong. Please try again." };
  if (replaced) await admin.from("documents").update({ status: "replaced" }).eq("id", replaced);
  refresh();
  return { ok: true };
}

/** The browser couldn't send the file: forget the half-finished upload. */
export async function cancelUpload(id: string) {
  const actor = await getActor();
  if (!actor || !uuid.test(id)) return;
  const { data: doc } = await actor.supabase.from("documents").select("file_key, status, uploaded_by").eq("id", id).maybeSingle();
  if (!doc || doc.status !== "uploading" || doc.uploaded_by !== actor.uid) return;
  await deleteObject(doc.file_key).catch(() => {});
  await createAdminClient().from("documents").delete().eq("id", id).eq("status", "uploading");
}

/** Remove a file uploaded by mistake, while it is still waiting for review. */
export async function deleteDocument(form: FormData) {
  const actor = await getActor();
  const id = String(form.get("id") ?? "");
  if (!actor || !uuid.test(id)) return;
  const { data: doc } = await actor.supabase.from("documents").select("file_key, status, uploaded_by").eq("id", id).maybeSingle();
  if (!doc || doc.status !== "under_review" || doc.uploaded_by !== actor.uid) return;
  await deleteObject(doc.file_key).catch(() => {});
  await createAdminClient().from("documents").delete().eq("id", id).eq("status", "under_review");
  refresh();
}

"use server";

import { refresh } from "next/cache";
import { cookies } from "next/headers";
import { getActor } from "@/lib/portal/actor";
import { STUDENT_COOKIE } from "@/lib/portal/context";
import { emails } from "@/lib/portal/email";
import { createAdminClient } from "@/lib/supabase/admin";
import { siteOrigin } from "@/lib/supabase/env";

export type LinkState = { error?: string; done?: string } | null;

const RELATIONSHIPS = ["Mother", "Father", "Parent", "Guardian"];
const str = (f: FormData, k: string) => String(f.get(k) ?? "").trim();

/** Parent: link by the student's email. Same answer whether or not the email has an account. */
export async function requestLinkByEmail(_: LinkState, form: FormData): Promise<LinkState> {
  const actor = await getActor();
  if (!actor || actor.profile.role !== "parent") return { error: "Only parent accounts can link to a student." };
  const email = str(form, "email").toLowerCase();
  const studentName = str(form, "name").slice(0, 120);
  const relationship = RELATIONSHIPS.includes(str(form, "relationship")) ? str(form, "relationship") : "Parent";
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email)) return { error: "Please enter your child's email address." };
  if (email === actor.profile.email.toLowerCase()) return { error: "Please enter your child's email, not your own." };
  if (!studentName) return { error: "Please enter your child's full name." };
  if (form.get("guardian") !== "on") return { error: "Please confirm you are the parent or legal guardian." };

  const admin = createAdminClient();
  const { data, error } = await admin.rpc("request_link_by_email", { p_parent: actor.uid, p_email: email, p_relationship: relationship });
  if (error) return { error: "Something went wrong. Please try again." };
  const result = data as { result: string; link_id?: string; student_id?: string };

  if (result.result === "rate_limited") return { error: "Too many requests. Please try again in an hour." };
  if (result.result === "already_linked") return { error: "You're already linked to this student." };
  if (result.result === "requested" && result.link_id) {
    await emails.linkRequest(email, actor.name);
  }
  if (result.result === "invite") {
    // No account yet: create the student's account and invite them to set a password.
    const { data: invited, error: inviteError } = await admin.auth.admin.inviteUserByEmail(email, {
      data: { role: "student", full_name: studentName, invited_by: actor.uid, inviter_name: actor.name },
      redirectTo: `${siteOrigin()}/auth/confirm`,
    });
    if (inviteError || !invited?.user) {
      console.error("[portal] invite failed:", inviteError?.message);
      return { error: "We couldn't send the invitation. Please try again later." };
    }
    const { error: linkError } = await admin.rpc("link_invited_student", {
      p_parent: actor.uid,
      p_student: invited.user.id,
      p_relationship: relationship,
      p_email: email,
    });
    if (linkError) {
      console.error("[portal] linking the invited student failed:", linkError.message);
      return { error: "We couldn't finish the request. Please try again later." };
    }
  }
  refresh();
  return { done: email };
}
