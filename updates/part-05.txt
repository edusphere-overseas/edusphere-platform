let gisScript: Promise<void> | null = null;
function loadGis() {
  gisScript ??= new Promise<void>((resolve, reject) => {
    if (window.google?.accounts?.id) return resolve();
    const s = document.createElement("script");
    s.src = "https://accounts.google.com/gsi/client";
    s.async = true;
    s.onload = () => resolve();
    s.onerror = () => {
      gisScript = null;
      reject(new Error("Google script blocked"));
    };
    document.head.appendChild(s);
  });
  return gisScript;
}

async function makeNonce() {
  const raw = btoa(String.fromCharCode(...crypto.getRandomValues(new Uint8Array(24)))).replace(/[+/=]/g, "");
  const hash = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(raw));
  return { raw, hashed: [...new Uint8Array(hash)].map((b) => b.toString(16).padStart(2, "0")).join("") };
}

/**
 * "Continue with Google". Uses Google's own button on this page, so Google's window names
 * the website (not the database address). Falls back to the redirect flow if Google's
 * script can't load (for example, blocked by a browser extension).
 */
function GoogleButton({ next, role, text }: { next?: string; role?: string; text: string }) {
  const box = useRef<HTMLDivElement>(null);
  const roleRef = useRef(role);
  const [mode, setMode] = useState<"loading" | "google" | "fallback">("loading");
  const [error, setError] = useState("");
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    roleRef.current = role;
  }, [role]);

  useEffect(() => {
    if (!googleOn) return;
    let cancelled = false;
    (async () => {
      try {
        await loadGis();
        const nonce = await makeNonce();
        if (cancelled || !box.current || !window.google) return;
        window.google.accounts.id.initialize({
          client_id: GOOGLE_CLIENT_ID,
          nonce: nonce.hashed,
          context: role ? "signup" : "signin",
          ux_mode: "popup",
          itp_support: true,
          use_fedcm_for_button: true,
          callback: async ({ credential }: { credential?: string }) => {
            if (!credential) return;
            setBusy(true);
            setError("");
            const res = await signInWithGoogleToken({ token: credential, nonce: nonce.raw, next, role: roleRef.current });
            if (res.to) window.location.assign(res.to);
            else {
              setBusy(false);
              setError(res.error ?? "Google sign-in didn't finish. Please try again.");
            }
          },
        });
        const width = Math.max(200, Math.min(400, Math.round(box.current.getBoundingClientRect().width || 320)));
        window.google.accounts.id.renderButton(box.current, {
          type: "standard",
          theme: "outline",
          size: "large",
          shape: "pill",
          text: role ? "signup_with" : "continue_with",
          logo_alignment: "center",
          width,
        });
        setMode("google");
      } catch {
        if (!cancelled) setMode("fallback");
      }
    })();
    return () => {
      cancelled = true;
    };
    // the button is rendered once; the chosen role is read when Google answers
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [next]);

  if (!googleOn) return null;
  if (mode === "fallback") {
    return (
      <form action={signInWithGoogle}>
        <input type="hidden" name="next" value={next ?? ""} />
        {role && <input type="hidden" name="role" value={role} />}

const one = (form: FormData, key: string, max: number) =>
  String(form.get(key) ?? "")
    .replace(/\s+/g, " ")
    .trim()
    .slice(0, max);

export async function readUniversityChoice(form: FormData): Promise<{ ok: true; value: UniversityChoice } | { ok: false; error: string; field: string }> {
  const country = one(form, "country", 40);
  if (!isCountry(country)) return { ok: false, error: "Please choose a country.", field: "country" };
  const ref = one(form, "university_ref", 40);
  const typed = one(form, "university", 160);
  const course = one(form, "course", 200);
  const intake = one(form, "intake", 40);

  let university = "";
  let universityRef: string | null = null;
  if (ref) {
    // The name always comes from the catalogue, never from the form
    const u = await getUniversity(ref);
    if (!u || u.country !== country) return { ok: false, error: "Please choose the university again.", field: "university" };
    university = u.name;
    universityRef = u.id;
  } else {
    if (typed.length < 2) return { ok: false, error: "Choose a university from the list, or type its name.", field: "university" };
    university = typed;
  }
  if (!isIntake(intake)) return { ok: false, error: "Please choose an intake.", field: "intake" };
  return { ok: true, value: { country, countryName: countryInfo(country)!.name, university, universityRef, course, intake } };
}

type Store = {
  generatedAt: string | null;
  list: Indexed<University>[];
  byId: Map<string, University>;
};
let store: Promise<Store> | null = null;
const courseStores = new Map<string, Promise<Indexed<Course>[]>>();

function loadStore() {
  store ??= (async () => {
    const raw = await readJson<{ generatedAt: string; items: RawUni[] }>("universities.json");
    const list: Indexed<University>[] = [];
    const byId = new Map<string, University>();
    for (const u of raw?.items ?? []) {
      if (!isCountry(u.c) || !u.n) continue;
      const item: University = {
        id: u.i,
        name: u.n,
        country: u.c,
        region: u.r ?? "",
        city: u.t ?? "",
        website: u.w ?? "",
        size: u.k ?? 0,
        fee: u.f ?? 0,
        currency: u.cur ?? "",
        pgwp: !!u.p,
        public: !!u.pub,
        courses: 0,
      };
      list.push({ item, hay: fold(`${item.name} ${item.city} ${item.region}`) });
      byId.set(item.id, item);
    }
    // Number of courses per university, for the countries that have course data
    await Promise.all(
      CATALOGUE_COUNTRIES.filter((c) => c.courses).map(async (c) => {
        for (const row of await loadCourses(c.id, byId)) row.item.university.courses += 1;
      }),
    );
    return { generatedAt: raw?.generatedAt ?? null, list, byId };
  })();
  return store;
}
