-- Access-rule tests for the portal. Run against a LOCAL database only:
--   psql "postgresql://postgres:postgres@127.0.0.1:54322/postgres" -v ON_ERROR_STOP=1 -f supabase/tests/access_rules.sql
-- Creates test accounts, checks what each role can and cannot do, then removes them.
\set QUIET on
set client_min_messages = notice;

drop schema if exists tests cascade;
delete from auth.users where email like 'test-%@example.test';
create schema tests;
grant usage on schema tests to authenticated, anon;

create table tests.ids (name text primary key, id uuid not null);
grant select on tests.ids to authenticated, anon;

create function tests.id(n text) returns uuid language sql stable as $$ select id from tests.ids where name = n $$;
-- the student's link code, read as the test runner (as if the student had shared it)
create function tests.code(n text) returns text language sql stable security definer set search_path = '' as $$
  select link_code from public.student_profiles where user_id = (select id from tests.ids where name = n) $$;

-- Act as a signed-in person for the rest of the transaction
create function tests.act_as(n text) returns void language plpgsql as $$
begin
  perform set_config('request.jwt.claims', json_build_object('sub', tests.id(n), 'role', 'authenticated')::text, true);
  perform set_config('role', 'authenticated', true);
end $$;

create function tests.ok(cond boolean, msg text) returns void language plpgsql as $$
begin
  if cond is not true then raise exception 'FAILED: %', msg; end if;
  raise notice 'ok  %', msg;
end $$;

create function tests.fails(stmt text, msg text) returns void language plpgsql as $$
begin
  begin
    execute stmt;
  exception when others then
    raise notice 'ok  % (refused: %)', msg, left(sqlerrm, 70);
    return;
  end;
  raise exception 'FAILED (was allowed): %', msg;
end $$;

create function tests.count_rows(stmt text) returns bigint language plpgsql as $$
declare n bigint;
begin
  execute 'select count(*) from (' || stmt || ') x' into n;
  return n;
end $$;

grant execute on all functions in schema tests to authenticated, anon;

  values ('00000000-0000-0000-0000-000000000000', uid, 'authenticated', 'authenticated', 'test-' || n || '@example.test', '', now(),
          '{"provider":"email","providers":["email"]}', meta, now(), now());
  insert into tests.ids values (n, uid);
  return uid;
end $$;
