  { id: "name", label: "Name (A to Z)" },
];
export type Facets = Record<"country" | "level" | "subject" | "location" | "fee" | "type", Record<string, number>>;
export type UniversityHit = { u: University; feeInr: number; matching: number };
export type CourseHit = { c: Course; feeInr: number };

type Dim = keyof Facets;
const DIMS: Dim[] = ["country", "level", "subject", "location", "fee", "type"];

function feeOk(inr: number, bucket: string) {
  const b = FEE_BUCKETS.find((x) => x.id === bucket);
  if (!b) return true;
  if (!inr) return false;
  return "max" in b ? inr <= b.max : inr >= b.min;
}
const feeBucketsOf = (inr: number) => (inr ? FEE_BUCKETS.filter((b) => feeOk(inr, b.id)).map((b) => b.id) : []);
const roundInr = (n: number) => (n ? Math.round(n / 1000) * 1000 : 0);

/** Which countries have course lists this month */
async function courseCountries() {
  const { byId } = await loadStore();
  const out = new Set<string>();
  for (const c of CATALOGUE_COUNTRIES.filter((x) => x.courses)) if ((await loadCourses(c.id, byId)).length) out.add(c.id);
  return out;
}

/** Drops filters that don't apply: degree and subject where there are no course lists, places of another country. */
export async function normaliseFilters(f: FinderFilters): Promise<FinderFilters & { courseData: boolean }> {
  const withCourses = await courseCountries();
  const courseData = f.country ? withCourses.has(f.country) : withCourses.size > 0;
  return {
    ...f,
    country: isCountry(f.country) ? f.country : "",
    levels: courseData ? f.levels.filter((l) => LEVELS.some((x) => x.id === l)) : [],
    subjects: courseData ? f.subjects.filter((s) => SUBJECTS.some((x) => x.id === s)) : [],
    locations: f.country ? f.locations : [],
    fee: FEE_BUCKETS.some((b) => b.id === f.fee) ? f.fee : "",
    types: f.types.filter((t) => t === "public" || t === "private"),
    courseData,
  };
}

type Row = { s: number; values: Record<Dim, string[]>; pass: Record<Dim, boolean> };

/** Counts for each filter option, given the other filters (so the counts say what a click would show). */
function facetCounts(rows: Row[]) {
  const facets = Object.fromEntries(DIMS.map((d) => [d, {}])) as Facets;
  for (const r of rows) {
    if (!r.s) continue;
    for (const d of DIMS) {
      if (!DIMS.every((o) => o === d || r.pass[o])) continue;
      for (const v of r.values[d]) facets[d][v] = (facets[d][v] ?? 0) + 1;
    }
  }
  return facets;
}

function universityRow(u: University, st: Stats | undefined, f: FinderFilters, inr: Record<string, number>, s: number) {
  const levelsFor = new Set<string>();
  const subjectsFor = new Set<string>();
  let matching = 0;
  for (const [key, n] of st?.combos ?? []) {
    const [l, sub] = key.split("|");
    const lOk = !f.levels.length || f.levels.includes(l);
    const sOk = !f.subjects.length || f.subjects.includes(sub);
    if (sOk) levelsFor.add(l);
    if (lOk) subjectsFor.add(sub);
    if (lOk && sOk) matching += n;
  }
  const native = (f.levels.length && Math.min(...f.levels.map((l) => st?.feeByLevel.get(l) || Infinity))) || u.feeYear;
  const feeInr = roundInr((Number.isFinite(native) ? native : u.feeYear) * (inr[u.currency] ?? 0));
  const courseFilter = f.levels.length || f.subjects.length;
  const row: Row = {
    s,
    values: {
      country: [u.country],
      level: [...levelsFor],
      subject: [...subjectsFor],
      location: u.location ? [u.location] : [],
      fee: feeBucketsOf(feeInr),
      type: u.type ? [u.type] : [],
    },
    pass: {
      country: !f.country || u.country === f.country,
      level: !f.levels.length || f.levels.some((l) => levelsFor.has(l)),
      subject: !f.subjects.length || f.subjects.some((x) => subjectsFor.has(x)),
      location: !f.locations.length || f.locations.includes(u.location),
      fee: feeOk(feeInr, f.fee),
      type: !f.types.length || f.types.includes(u.type),
    },
  };
  return { row, hit: { u, feeInr, matching: courseFilter ? matching : u.courses } };
}

function sortHits<T extends { feeInr: number }>(hits: { h: T; s: number; size: number; name: string }[], sort: FinderSort) {
  const known = (a: number) => (a ? 0 : 1);
  hits.sort((a, b) => {
    if (sort === "fee-asc") return known(a.h.feeInr) - known(b.h.feeInr) || a.h.feeInr - b.h.feeInr || b.size - a.size;
    if (sort === "fee-desc") return known(a.h.feeInr) - known(b.h.feeInr) || b.h.feeInr - a.h.feeInr || b.size - a.size;
    if (sort === "name") return a.name.localeCompare(b.name);
    return b.s - a.s || b.size - a.size || a.name.localeCompare(b.name);
  });
}

export async function listUniversities(f: FinderFilters, { sort = "recommended", offset = 0, limit = 20 }: { sort?: FinderSort; offset?: number; limit?: number } = {}) {
  const { list, stats, inr } = await loadStore();
  const words = tokens(f.q);
  const phrase = words.join(" ");
  const rows: Row[] = [];
  const hits: { h: UniversityHit; s: number; size: number; name: string }[] = [];
  for (const { item: u, hay } of list) {
    const s = score(hay, fold(u.name), words, phrase, false);
    if (!s) continue;
    const { row, hit } = universityRow(u, stats.get(u.id), f, inr, s);
    rows.push(row);
    if (DIMS.every((d) => row.pass[d])) hits.push({ h: hit, s, size: u.size, name: u.name });
  }
  sortHits(hits, sort);
  return { total: hits.length, items: hits.slice(offset, offset + limit).map((x) => x.h), facets: facetCounts(rows) };
}

export async function listCourses(f: FinderFilters, { sort = "recommended", offset = 0, limit = 20 }: { sort?: FinderSort; offset?: number; limit?: number } = {}) {
  const { byId, inr } = await loadStore();
  const words = tokens(f.q);
  const phrase = words.join(" ");
  const rows: Row[] = [];
  const hits: { h: CourseHit; s: number; size: number; name: string }[] = [];
  for (const ct of CATALOGUE_COUNTRIES.filter((c) => c.courses)) {
    for (const { item: c, hay } of await loadCourses(ct.id, byId)) {
      if (f.university && c.university.id !== f.university) continue;
      const s = score(hay, fold(c.title), words, phrase);
      if (!s) continue;
      const u = c.university;
      const feeInr = roundInr((firstYear(c) || u.feeYear) * (inr[u.currency] ?? 0));
      const row: Row = {
        s,
        values: { country: [u.country], level: [c.level], subject: [c.subject], location: u.location ? [u.location] : [], fee: feeBucketsOf(feeInr), type: u.type ? [u.type] : [] },
        pass: {
          country: !f.country || u.country === f.country,
          level: !f.levels.length || f.levels.includes(c.level),
          subject: !f.subjects.length || f.subjects.includes(c.subject),
          location: !f.locations.length || f.locations.includes(u.location),
          fee: feeOk(feeInr, f.fee),
          type: !f.types.length || f.types.includes(u.type),
        },
      };
      rows.push(row);
      if (DIMS.every((d) => row.pass[d])) hits.push({ h: { c, feeInr }, s, size: u.size, name: `${c.title} ${u.name}` });
    }
  }
  sortHits(hits, sort);
  return { total: hits.length, items: hits.slice(offset, offset + limit).map((x) => x.h), facets: facetCounts(rows) };
}

  return `${years} year${years === 1 ? "" : "s"}`;
}

export async function universityDetails(id: string) {
  "use cache";
  cacheLife("days");
  const { byId, stats, inr } = await loadStore();
  const u = byId.get(id);
  if (!u) return null;
  const courses = u.courses ? (await loadCourses(u.country, byId)).filter((x) => x.item.university.id === id).map((x) => x.item) : [];
  const levels = LEVELS.filter((l) => courses.some((c) => c.level === l.id)).map((l) => l.id);
  const st = stats.get(id);
  const feeInr = roundInr(u.feeYear * (inr[u.currency] ?? 0));
  const feeByLevel = Object.fromEntries([...(st?.feeByLevel ?? [])].filter(([l]) => l !== "*").map(([l, v]) => [l, roundInr(v * (inr[u.currency] ?? 0))]));
  const base: FinderFilters = { q: "", country: u.country, levels: [], subjects: [], locations: [], fee: "", types: [] };
  let similar = u.location ? (await listUniversities({ ...base, locations: [u.location] }, { limit: 5 })).items.filter((h) => h.u.id !== id) : [];
  if (similar.length < 2) similar = (await listUniversities(base, { limit: 5 })).items.filter((h) => h.u.id !== id);
  similar = similar.slice(0, 4);
  return {
    university: u,
    feeInr,
    feeByLevel,
    levels,
    rate: inr[u.currency] ?? 0,
    courses: courses
      .sort((a, b) => LEVELS.findIndex((l) => l.id === a.level) - LEVELS.findIndex((l) => l.id === b.level) || a.title.localeCompare(b.title))
      .slice(0, 600)
      .map((c) => ({ title: c.title, level: c.level, subject: c.subject, months: c.months, fee: c.fee, feeYearInr: roundInr(firstYear(c) * (inr[u.currency] ?? 0)), url: c.url })),
    similar,
  };
}

  return {
    academic: `${needs[0].toUpperCase()}${needs.slice(1)}, usually ${r.min}% or more (${r.competitive}%+ for the better-known universities)`,
    english: `IELTS ${r.ielts.toFixed(1)} overall, no band below ${r.band.toFixed(1)} (or PTE ${toPte(r.ielts)}, TOEFL ${toToefl(r.ielts)}, Duolingo ${toDuolingo(r.ielts)})`,
    notes: r.notes ?? [],
  };
}

/** Rough conversions to a percentage (CBSE's ×9.5 for a 10-point CGPA) */
export function toPercent(score: number, scale: string) {
  if (!(score > 0)) return 0;
  if (scale === "cgpa10") return Math.min(100, score * 9.5);
  if (scale === "cgpa4") return Math.min(100, score * 25);
  return Math.min(100, score);
}

/** Approximate IELTS equivalents, from the test makers' published comparison tables */
const PTE = [[36, 5.0], [42, 5.5], [50, 6.0], [58, 6.5], [65, 7.0], [73, 7.5], [79, 8.0], [84, 8.5]] as const;
const TOEFL = [[35, 5.0], [46, 5.5], [60, 6.0], [79, 6.5], [94, 7.0], [102, 7.5], [110, 8.0], [115, 8.5]] as const;
const DUOLINGO = [[75, 5.0], [85, 5.5], [95, 6.0], [105, 6.5], [120, 7.0], [130, 7.5], [140, 8.0], [150, 8.5]] as const;
const fromTable = (score: number, table: readonly (readonly [number, number])[]) => {
  let out = 0;
  for (const [min, band] of table) if (score >= min) out = band;
  return out;
};
const toTable = (ielts: number, table: readonly (readonly [number, number])[]) => table.find(([, band]) => band >= ielts)?.[0] ?? table[table.length - 1][0];
const toPte = (ielts: number) => toTable(ielts, PTE);
const toToefl = (ielts: number) => toTable(ielts, TOEFL);
const toDuolingo = (ielts: number) => toTable(ielts, DUOLINGO);

export function toIelts(test: string, score: number) {
  if (!(score > 0)) return 0;
  if (test === "ielts") return Math.min(9, score);
  if (test === "pte") return fromTable(score, PTE);
  if (test === "toefl") return fromTable(score, TOEFL);
  if (test === "duolingo") return fromTable(score, DUOLINGO);
  return 0;
}

export type EligibilityInput = {
  country: string;
  level: string;
  qualification: string;
  score: number;
  scale: string;
  passingYear: number;
  test: string;
  testScore: number;
  backlogs: number;
};

export type Verdict = "strong" | "good" | "possible" | "pathway" | "not_yet";
export type EligibilityResult = {
  verdict: Verdict;
  headline: string;
  summary: string;
  points: { ok: boolean | null; text: string }[];
  /** a better-fitting level to suggest, when the chosen one doesn't fit */
  suggestLevel?: string;
};

const LEVEL_NAMES: Record<string, string> = {
  foundation: "a foundation programme",
  diploma: "a diploma",
  bachelor: "a bachelor's degree",
  pgcert: "a postgraduate diploma",
  master: "a master's degree",
  doctorate: "a PhD",
};

export function checkEligibilityRules(input: EligibilityInput, thisYear: number): EligibilityResult | null {
  const r = requirements(input.country, input.level);
  if (!r) return null;
  const pct = Math.round(toPercent(input.score, input.scale) * 10) / 10;
  const ielts = input.test === "none" ? 0 : toIelts(input.test, input.testScore);
  const points: EligibilityResult["points"] = [];
  const wants = LEVEL_NAMES[input.level] ?? "this level";

  // Wrong starting point: e.g. a master's straight after the 12th
  if (!r.needs.includes(input.qualification as QualificationId)) {
    const suggest = r.needs.includes("master") && !r.needs.includes("bachelor3") ? (input.qualification === "12th" || input.qualification === "diploma" ? "bachelor" : "master") : "bachelor";
    return {
      verdict: "not_yet",
      headline: `${wants[0].toUpperCase()}${wants.slice(1)} needs a different qualification`,
      summary: `For ${wants} you usually need ${r.needs.includes("12th") ? "the 12th" : r.needs.length === 1 ? "a master's degree" : "a bachelor's degree"} first. Your counsellor can suggest the right route.`,
      points: [{ ok: false, text: `Your highest qualification: ${qualificationLabel(input.qualification)}` }],
      suggestLevel: suggest !== input.level ? suggest : undefined,
    };
  }

  if (pct >= r.competitive) points.push({ ok: true, text: `Your ${pct}% meets what the better-known universities usually ask for (${r.competitive}%+).` });
  else if (pct >= r.min) points.push({ ok: true, text: `Your ${pct}% meets the usual minimum (${r.min}%). Some top universities ask for ${r.competitive}% or more.` });
  else if (pct >= r.min - 5) points.push({ ok: null, text: `Your ${pct}% is just under the usual minimum (${r.min}%). Some universities accept it, especially with good English scores or work experience.` });
  else points.push({ ok: false, text: `Your ${pct}% is under the usual minimum (${r.min}%).` });

  if (!ielts) points.push({ ok: null, text: `You'll need an English test: usually IELTS ${r.ielts.toFixed(1)} overall (or PTE ${toPte(r.ielts)}). We can help you prepare.` });
  else if (ielts >= r.ielts) points.push({ ok: true, text: `Your English score (about IELTS ${ielts.toFixed(1)}) meets the usual ${r.ielts.toFixed(1)}. Check that no band is below ${r.band.toFixed(1)}.` });
  else if (ielts >= r.ielts - 0.5) points.push({ ok: null, text: `Your English score (about IELTS ${ielts.toFixed(1)}) is just under the usual ${r.ielts.toFixed(1)}. Some universities accept it, or offer a short English course first.` });
  else points.push({ ok: false, text: `Your English score (about IELTS ${ielts.toFixed(1)}) is under the usual ${r.ielts.toFixed(1)}. A retake or a pre-sessional English course would help.` });

  const pg = ["pgcert", "master", "doctorate"].includes(input.level);
  if (input.backlogs > 0) {
    const limit = input.country === "uk" ? 10 : input.country === "usa" ? 8 : 5;
    points.push(
      input.backlogs <= limit
        ? { ok: true, text: `${input.backlogs} backlog${input.backlogs === 1 ? "" : "s"}: accepted by many universities in ${countryShort(input.country)}.` }
        : { ok: null, text: `${input.backlogs} backlogs: some universities limit backlogs, so we'll shortlist ones that accept your record.` },
    );
  }

  const gap = input.passingYear ? thisYear - input.passingYear : 0;
  if (gap > (pg ? 5 : 2)) points.push({ ok: null, text: `A ${gap}-year gap since you finished: work experience or other activity helps explain it.` });
  if (input.country === "usa" && input.level === "master" && input.qualification === "bachelor3")
    points.push({ ok: null, text: "A 3-year bachelor's: some US universities accept it, others ask for 16 years of education." });

  const academic = pct >= r.competitive ? 2 : pct >= r.min ? 1 : pct >= r.min - 5 ? 0 : -1;
  const english = !ielts ? 1 : ielts >= r.ielts ? 2 : ielts >= r.ielts - 0.5 ? 0 : -1;
  let verdict: Verdict;
  if (academic === 2 && english >= 1) verdict = "strong";
  else if (academic >= 1 && english >= 0) verdict = "good";
  else if (academic >= 0) verdict = "possible";
  else verdict = "pathway";

  const where = countryShort(input.country);
  const copy: Record<Verdict, [string, string]> = {
    strong: [`You look a strong fit for ${wants} in ${where}`, "Your marks fit a wide range of universities, including better-known ones."],
    good: [`You look eligible for ${wants} in ${where}`, "Your profile fits the usual requirements at many universities."],
    possible: [`You could be eligible for ${wants} in ${where}`, "Some universities would consider your profile. Your counsellor can shortlist the ones that fit."],
    pathway: [`${where} is possible with the right route`, "A foundation, diploma or pathway programme can lead to the degree you want."],
    not_yet: ["", ""],
  };
  return {
    verdict,
    headline: copy[verdict][0],
    summary: `${copy[verdict][1]}${!ielts ? " You'll also need an English test score." : ""}`,
    points: [...points, ...(r.notes ?? []).map((text) => ({ ok: null, text }))],
    suggestLevel: verdict === "pathway" ? (pg ? "pgcert" : "foundation") : undefined,
  };
}

export function ageFrom(dob: string | null | undefined) {
  if (!dob) return null;
  const d = new Date(dob + "T00:00:00Z");
  const now = new Date();
  let age = now.getUTCFullYear() - d.getUTCFullYear();
  const m = now.getUTCMonth() - d.getUTCMonth();
  if (m < 0 || (m === 0 && now.getUTCDate() < d.getUTCDate())) age--;
  return age;
}
