/** A student (or their parent) adds a university to the student's application tracker, as a shortlist entry. */
export async function addUniversity(_: AddUniversityState, form: FormData): Promise<AddUniversityState> {
  const p = await getPortal();
  if (p.staff) return { error: "Staff add universities from the student's page in the staff dashboard." };
  if (!p.student) return { error: "Link to a student first, then add universities to their list." };
  const choice = await readUniversityChoice(form);
  if (!choice.ok) return { error: choice.error, field: choice.field };
  const c = choice.value;

  const { data: existing } = await p.supabase.from("applications").select("id, university, course, status").eq("student_id", p.student.id);
  if ((existing ?? []).length >= 30) return { error: "Your list is full. Ask your counsellor to tidy it up first." };
  const same = (existing ?? []).find((a) => a.university.toLowerCase() === c.university.toLowerCase() && (a.course ?? "").toLowerCase() === c.course.toLowerCase());
  if (same) return { error: "This university and course are already on your list.", field: "university" };

  const { data, error } = await p.supabase
    .from("applications")
    .insert({
      student_id: p.student.id,
      university: c.university,
      university_ref: c.universityRef,
      country: c.country,
      course: c.course,
      intake: c.intake || null,
      stage: "shortlist",
      status: "active",
      created_by: p.userId,
    })
    .select("id")
    .single();
  if (error || !data) {
    console.error("[portal] add university failed:", error?.code, error?.message);
    return { error: error?.code === "42501" ? "You can't add universities for this student." : "Something went wrong. Please try again." };
  }

  const contacts = await familyContacts(p.student.id);
  const to = contacts.counsellorEmail ?? staffInbox();
  if (to) {
    const who = p.role === "parent" ? `${p.profile.full_name || p.profile.email} (parent of ${contacts.studentName})` : contacts.studentName;
    await sendEmail({
      to,
      subject: `${contacts.studentName} added ${c.university} to their list`,
      heading: "A university was added in the portal",
      lines: [`${who} added this to the application tracker as a shortlist entry:`, describeChoice(c), "Move it to Applied once the application is sent."],
      button: { label: "Open the student", path: `/staff/students/${p.student.id}` },
    });
  }
  redirect(`/portal/applications?added=${data.id}`);
}

/** Remove a university the person added themselves, while it is still on the shortlist. */
export async function removeUniversity(form: FormData) {
  const p = await getPortal();

  if (!uuid.test(id)) return;
  await p.supabase.from("applications").delete().eq("id", id).eq("created_by", p.userId).eq("stage", "shortlist");
  refresh();
}

function UniversitySearch({ country, invalid, onPick, onNotListed }: { country: string; invalid?: boolean; onPick: (u: NonNullable<PickedUniversity>) => void; onNotListed: () => void }) {
  const [q, setQ] = useState("");
  const [items, setItems] = useState<Option[]>([]);
  const [open, setOpen] = useState(false);
  const [busy, setBusy] = useState(false);
  const [active, setActive] = useState(-1);
  const box = useRef<HTMLDivElement>(null);
  const ids = useId();
  const listId = `${ids}-list`;

  useEffect(() => {
    if (!country) return;
    const ctrl = new AbortController();
    const t = setTimeout(async () => {
      setBusy(true);
      try {
        const res = await fetch(`/api/catalogue?kind=universities&country=${encodeURIComponent(country)}&q=${encodeURIComponent(q)}`, { signal: ctrl.signal });
        const json = (await res.json()) as { items: Option[] };
        setItems(json.items ?? []);
        setActive(-1);
      } catch {
        /* aborted or offline: keep the last list */
      } finally {
        setBusy(false);
      }
    }, 180);
    return () => {
      clearTimeout(t);
      ctrl.abort();
    };
  }, [q, country]);

  useEffect(() => {
    const close = (e: MouseEvent) => {
      if (box.current && !box.current.contains(e.target as Node)) setOpen(false);
    };
    document.addEventListener("mousedown", close);
    return () => document.removeEventListener("mousedown", close);
  }, []);

  const total = items.length + 1; // + "not listed"
  const choose = (i: number) => {
    if (i === items.length) onNotListed();
    else if (items[i]) onPick({ id: items[i].id, name: items[i].name, place: items[i].place });
  };

  return (
    <div ref={box} className={`${label} relative`}>
      <label htmlFor={`${ids}-q`}>University</label>
      <span className="relative">
        <Search className="pointer-events-none absolute top-1/2 left-3.5 h-4.5 w-4.5 -translate-y-1/2 text-slate/70" aria-hidden="true" />
        <input
          id={`${ids}-q`}
          type="text"
          role="combobox"
          aria-expanded={open && !!country}
          aria-controls={listId}
          aria-autocomplete="list"
          aria-activedescendant={active >= 0 ? `${ids}-o${active}` : undefined}
          aria-invalid={invalid || undefined}
          disabled={!country}
          value={q}
          placeholder={country ? "Start typing a university name" : "Choose a country first"}
          autoComplete="off"
          onChange={(e) => {
            setQ(e.target.value);
            setOpen(true);
          }}
          onFocus={() => setOpen(true)}
          onKeyDown={(e) => {
            if (e.key === "ArrowDown") {
              e.preventDefault();
              setOpen(true);
              setActive((a) => (a + 1) % total);
            } else if (e.key === "ArrowUp") {
              e.preventDefault();
              setActive((a) => (a <= 0 ? total - 1 : a - 1));
            } else if (e.key === "Enter" && open && active >= 0) {
              e.preventDefault();
              choose(active);
            } else if (e.key === "Escape") setOpen(false);
          }}
          className={`${input} pl-11`}
        />
        {busy && <Loader2 className="absolute top-1/2 right-3.5 h-4 w-4 -translate-y-1/2 animate-spin text-slate" aria-hidden="true" />}
      </span>
      {open && country && (
        <ul id={listId} role="listbox" className="absolute top-full right-0 left-0 z-30 mt-1.5 max-h-80 overflow-auto rounded-xl bg-white p-1.5 font-normal shadow-[0_18px_40px_-12px_rgba(13,34,54,0.35)] ring-1 ring-deep/10">
          {items.map((u, i) => (
            <li
              key={u.id}
              id={`${ids}-o${i}`}
              role="option"
              aria-selected={active === i}
              onMouseDown={(e) => e.preventDefault()}
              onClick={() => choose(i)}

            </li>
          ))}
          {!busy && items.length === 0 && <li className="px-3 py-2.5 text-sm text-slate">No match in the list.</li>}
          <li
            id={`${ids}-o${items.length}`}
            role="option"
            aria-selected={active === items.length}

            onClick={() => choose(items.length)}
            className={`mt-1 flex cursor-pointer items-center gap-2 rounded-lg border-t border-deep/8 px-3 py-2.5 text-sm font-semibold text-teal ${active === items.length ? "bg-mist" : "hover:bg-mist"}`}

            My university isn&apos;t listed: type its name
