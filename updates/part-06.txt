/** One phone number, WhatsApp number or email address, with a label. */
export const contactPoint = defineType({
  name: "contactPoint",
  title: "Contact detail",
  type: "object",
  fields: [
    defineField({
      name: "kind",
      title: "Type",
      type: "string",
      options: {
        list: [
          { title: "Phone", value: "phone" },
          { title: "WhatsApp", value: "whatsapp" },
          { title: "Email", value: "email" },
        ],
        layout: "radio",
        direction: "horizontal",
      },
      initialValue: "phone",
      validation: (r) => r.required(),
    }),
    defineField({
      name: "label",
      title: "Label",
      type: "string",
      description: 'Shown next to the number, e.g. "Admissions", "UK desk", "General enquiries".',
      validation: (r) => r.required().max(40),
    }),
    defineField({
      name: "value",
      title: "Number or email",
      type: "string",
      description: "Phone and WhatsApp: include the country code, e.g. +91 81294 40282.",
      validation: (r) =>
        r.required().custom((value, ctx) => {
          const kind = (ctx.parent as { kind?: string })?.kind;
          if (!value) return true;
          if (kind === "email") return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value) || "Enter a valid email address";
          return /^\+\d[\d\s-]{6,}$/.test(value) || "Start with + and the country code, e.g. +91 81294 40282";
        }),
    }),
    defineField({ name: "primary", title: "Main contact of this type", type: "boolean", initialValue: false }),
    defineField({ name: "show", title: "Show on website", type: "boolean", initialValue: true }),
  ],
  preview: {
    select: { kind: "kind", label: "label", value: "value", show: "show" },
    prepare: ({ kind, label, value, show }) => ({
      title: `${label ?? ""}: ${value ?? ""}`,
      subtitle: `${kind === "email" ? "Email" : kind === "whatsapp" ? "WhatsApp" : "Phone"}${show === false ? " · hidden" : ""}`,
    }),
  },
});

import type { StructureResolver } from "sanity/structure";
import { Home, Settings } from "lucide-react";

const ordered = (S: Parameters<StructureResolver>[0], type: string, title: string) =>
  S.listItem()
    .title(title)
    .schemaType(type)
    .child(S.documentTypeList(type).title(title).defaultOrdering([{ field: "order", direction: "asc" }]));

/** The left-hand menu of the Content area, in plain words for the team. */
export const structure: StructureResolver = (S) =>
  S.list()
    .title("Content")
    .items([
      S.listItem()
        .title("Website settings")
        .id("siteSettings")
        .icon(Settings)
        .child(S.document().schemaType("siteSettings").documentId("siteSettings").title("Website settings")),
      S.divider(),
      S.listItem()
        .title("Home page")
        .icon(Home)
        .child(
          S.list()
            .title("Home page")
            .items([
              ordered(S, "banner", "Banners"),
              ordered(S, "tickerItem", "News ticker"),
              ordered(S, "destination", "Destinations"),
              ordered(S, "service", "Student services"),
              ordered(S, "onshoreProgramme", "UK onshore programmes"),
            ]),
        ),
      S.listItem()
        .title('"Open now" visas and scholarships')
        .schemaType("openNowProgramme")
        .child(
          S.documentTypeList("openNowProgramme")
            .title('"Open now"')
            .defaultOrdering([{ field: "closesAt", direction: "asc" }]),
        ),
      S.divider(),
      ordered(S, "office", "Offices"),
      ordered(S, "teamMember", "Team"),
    ]);
