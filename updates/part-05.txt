type QueryResult = {
  settings: {
    companyName?: string;
    legalName?: string;
    tagline?: string;
    description?: string;
    whatsappNumber?: string;
    whatsappMessage?: string;
    contacts?: ContactPoint[] | null;
    socialLinks?: SocialLink[] | null;
    heroEyebrow?: string;
    heroTitleLine1?: string;
    heroTitleLine2?: string;
    heroIntro?: string;
    heroNote?: string;
    sections?: Partial<Sections> | null;
  } | null;
  offices: (Omit<Office, "id" | "contacts" | "photos" | "mapQuery"> & {
    _id: string;
    mapQuery?: string | null;
    contacts?: ContactPoint[] | null;
    photos?: SanityPhoto[] | null;
  })[];
  ticker: SiteContent["ticker"];
  banners: (Omit<Banner, "id"> & { _id: string })[];
  destinations: SiteContent["destinations"];
  services: SiteContent["services"];
  openNow: OpenNowItem[];
  onshore: SiteContent["onshore"];
  team: SiteContent["team"];
};

/**
 * All site content. Comes from the Content area (Sanity) once "Website settings"
 * exists there; until then the built-in defaults keep the site complete.
 */
export async function getSiteContent(): Promise<SiteContent> {
  const data = await sanityQuery<QueryResult>(SITE_QUERY);
  if (!data?.settings) return defaults();

  const fallback = defaults();
  const s = data.settings;
  return {
    fromCms: true,
    companyName: s.companyName || fallback.companyName,
    legalName: s.legalName || s.companyName || fallback.legalName,
    tagline: s.tagline || fallback.tagline,
    description: s.description || fallback.description,
    whatsappNumber: s.whatsappNumber ? digits(s.whatsappNumber) : fallback.whatsappNumber,
    whatsappMessage: s.whatsappMessage || fallback.whatsappMessage,
    contacts: s.contacts ?? [],
    socialLinks: (s.socialLinks ?? []).filter((l) => l.url),
    hero: {
      eyebrow: s.heroEyebrow ?? fallback.hero.eyebrow,
      titleLine1: s.heroTitleLine1 || fallback.hero.titleLine1,
      titleLine2: s.heroTitleLine2 ?? fallback.hero.titleLine2,
      intro: s.heroIntro ?? fallback.hero.intro,
      note: s.heroNote ?? fallback.hero.note,
    },
    sections: { ...defaultSections, ...(s.sections ?? {}) },
    offices: (data.offices ?? []).map((o) => ({
      ...o,
      id: o._id,
      mapQuery: o.mapQuery || [o.legalName, ...o.addressLines].filter(Boolean).join(", "),
      contacts: o.contacts ?? [],
      photos: (o.photos ?? []).filter((p) => p?.asset?.url),
    })),
    ticker: data.ticker ?? [],
    banners: (data.banners ?? []).map((b) => ({ ...b, id: b._id, image: b.image?.asset?.url ? b.image : null })),
    destinations: (data.destinations ?? []).filter((d) => d.id),
    services: data.services ?? [],
    openNow: data.openNow ?? [],
    onshore: data.onshore ?? [],
    team: (data.team ?? []).map((t) => ({ ...t, photo: t.photo?.asset?.url ? t.photo : null })),
  };
}

import { cacheLife, cacheTag } from "next/cache";
import { client } from "./client";
import { SANITY_TAG } from "./tags";

/**
 * Cached Sanity query. Results are shared by every visitor and refreshed about
 * once a minute, or immediately when the Sanity webhook calls /api/revalidate.
 * Returns null if Sanity can't be reached, so the site falls back to defaults.
 */
export async function sanityQuery<T>(query: string, params: Record<string, unknown> = {}): Promise<T | null> {
  "use cache";
  cacheTag(SANITY_TAG);
  cacheLife("minutes");
  try {
    return await client.fetch<T>(query, params);
  } catch (error) {
    console.error("[sanity] query failed:", error instanceof Error ? error.message : error);
    return null;
  }
}

import { defineQuery } from "next-sanity";

const scheduled = `active != false && (!defined(startsAt) || startsAt <= now()) && (!defined(endsAt) || endsAt > now())`;
const photoFields = `{ ..., "alt": coalesce(alt, ""), "caption": caption, asset->{ _id, url, metadata { dimensions, lqip } } }`;
const contact = `{ kind, label, value, primary }`;

/** Everything the site layout and home page need, in one request. */
export const SITE_QUERY = defineQuery(`{
  "settings": *[_id == "siteSettings"][0]{
    companyName, legalName, tagline, description, whatsappNumber, whatsappMessage,
    "contacts": contacts[show != false]${contact},
    "socialLinks": socialLinks[show != false]{ platform, handle, url },
    heroEyebrow, heroTitleLine1, heroTitleLine2, heroIntro, heroNote,
    sections
  },
  "offices": *[_type == "office" && showOnWebsite == true && count(addressLines) > 0] | order(isHeadOffice desc, order asc){
    _id, name, legalName, isHeadOffice, country, countryCode, addressLines, mapQuery, googleBusinessUrl, timeZone,
    "contacts": contacts[show != false]${contact},
    "photos": photos[]${photoFields}
  },
  "ticker": *[_type == "tickerItem" && ${scheduled}] | order(order asc){ tag, text, link },
  "banners": *[_type == "banner" && ${scheduled}] | order(order asc){
    _id, eyebrow, title, subtitle, flag, ctaLabel, ctaLink, "image": image${photoFields}
  },
  "destinations": *[_type == "destination" && show != false] | order(order asc){
    "id": slug.current, name, flag, summary, intakes
  },
  "services": *[_type == "service" && show != false] | order(order asc){ title, description, icon },
  "openNow": *[_type == "openNowProgramme" && show == true && (!defined(closesAt) || closesAt > now())] | order(coalesce(closesAt, "9999") asc, order asc){
    _id, title, kind, country, flag, summary, details, opensAt, closesAt, officialLink, lastChecked, ctaLabel
  },
  "onshore": *[_type == "onshoreProgramme" && show != false] | order(order asc){ _id, title, provider, location, summary, intakes },
  "team": *[_type == "teamMember" && show != false] | order(order asc){
    _id, name, role, bio, languages, "office": office->name, "photo": photo${photoFields}
  }
}`);
