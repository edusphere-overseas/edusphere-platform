export default async function RootLayout({ children }: LayoutProps<"/">) {
  // Logo intro length and replay rule come from Content → Website settings → Logo intro.
  const { intro } = await getSiteContent();

        <script dangerouslySetInnerHTML={{ __html: introScript(intro) }} />
      </head>
      <body className="min-h-dvh antialiased">{children}</body>
    </html>
  );
}

import type { MetadataRoute } from "next";
import { site } from "@/config/site";

/**
 * Search engines are kept out until launch (Step 7). At launch, set
 * SITE_LAUNCHED=true in Netlify and remove `robots` noindex in app/layout.tsx.
 */
export default function robots(): MetadataRoute.Robots {
  if (process.env.SITE_LAUNCHED !== "true") return { rules: { userAgent: "*", disallow: "/" } };
  return {
    rules: { userAgent: "*", allow: "/", disallow: ["/admin", "/portal", "/api", "/__forms.html"] },
    sitemap: `${site.url}/sitemap.xml`,
    host: site.url,
  };
}

import type { MetadataRoute } from "next";
import { site } from "@/config/site";
import { getLibrary, getSiteContent } from "@/lib/content";

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const [c, lib] = await Promise.all([getSiteContent(), getLibrary()]);
  const pages = [
    "",
    "/destinations",
    "/services",
    "/universities",
    "/scholarships",
    "/success-stories",
    "/news",
    "/faqs",
    "/about",
    "/contact",
    ...(c.sections.courseFinder ? ["/course-finder"] : []),
  ];
  return [
    ...pages.map((p) => ({ url: `${site.url}${p}`, changeFrequency: "weekly" as const, priority: p === "" ? 1 : 0.7 })),
    ...lib.destinations.map((d) => ({ url: `${site.url}/destinations/${d.id}`, changeFrequency: "monthly" as const, priority: 0.8 })),
    ...lib.posts.map((p) => ({ url: `${site.url}/news/${p.slug}`, lastModified: p.date, changeFrequency: "yearly" as const, priority: 0.5 })),
  ];
}
