/** The university a student picked (from the catalogue, or typed in), read from a form. */
export type UniversityChoice = {
  country: string;
  countryName: string;
  university: string;
  universityRef: string | null;
  course: string;
  intake: string;
};

const one = (form: FormData, key: string, max: number) =>
  String(form.get(key) ?? "")
    .replace(/\s+/g, " ")
    .trim()
    .slice(0, max);

export async function readUniversityChoice(form: FormData): Promise<{ ok: true; value: UniversityChoice } | { ok: false; error: string; field: string }> {
  const country = one(form, "country", 40);
  if (!isCountry(country)) return { ok: false, error: "Please choose a country.", field: "country" };
  const ref = one(form, "university_ref", 40);
  const typed = one(form, "university", 160);
  const course = one(form, "course", 200);
  const intake = one(form, "intake", 40);

  let university = "";
  let universityRef: string | null = null;
  if (ref) {
    // The name always comes from the catalogue, never from the form
    const u = await getUniversity(ref);
    if (!u || u.country !== country) return { ok: false, error: "Please choose the university again.", field: "university" };
    university = u.name;
    universityRef = u.id;
  } else {
    if (typed.length < 2) return { ok: false, error: "Choose a university from the list, or type its name.", field: "university" };
    university = typed;
  }
  if (!isIntake(intake)) return { ok: false, error: "Please choose an intake.", field: "intake" };
  return { ok: true, value: { country, countryName: countryInfo(country)!.name, university, universityRef, course, intake } };
}

export const describeChoice = (c: UniversityChoice) =>
  [c.university, c.course, c.countryName, c.intake && `${c.intake} intake`].filter(Boolean).join(" · ");

/** The study level and education details from the eligibility and apply forms (all optional here). */
export type Education = {
  level: string;
  qualification: string;
  score: number;
  scale: string;
  passingYear: number;
  test: string;
  testScore: number;
  backlogs: number;
  city: string;
};

const num = (form: FormData, key: string) => {
  const n = Number.parseFloat(one(form, key, 12));
  return Number.isFinite(n) ? n : 0;
};

export function readEducation(form: FormData, thisYear: number): { ok: true; value: Education } | { ok: false; error: string; field: string } {
  const level = one(form, "level", 20);
  const qualification = one(form, "qualification", 20);
  const scale = one(form, "scale", 10) || "percent";
  const test = one(form, "test", 10) || "none";
  const value: Education = {
    level: LEVELS.some((l) => l.id === level) ? level : "",
    qualification: QUALIFICATIONS.some((q) => q.id === qualification) ? qualification : "",
    score: num(form, "score"),
    scale: SCORE_SCALES.some((s) => s.id === scale) ? scale : "percent",
    passingYear: Math.round(num(form, "passing_year")),
    test: ENGLISH_TESTS.some((t) => t.id === test) ? test : "none",
    testScore: num(form, "test_score"),
    backlogs: Math.max(0, Math.min(99, Math.round(num(form, "backlogs")))),
    city: one(form, "city", 80),
  };
  const max = SCORE_SCALES.find((s) => s.id === value.scale)!.max;
  if (value.score < 0 || value.score > max) return { ok: false, error: `Please enter your marks (up to ${max}).`, field: "score" };
  const testMax = ENGLISH_TESTS.find((t) => t.id === value.test)!.max;
  if (value.test !== "none" && (value.testScore < 0 || value.testScore > testMax)) return { ok: false, error: `Please check your ${englishLabel(value.test)} score.`, field: "test_score" };
  if (value.passingYear && (value.passingYear < 1970 || value.passingYear > thisYear + 2)) return { ok: false, error: "Please check the year you finished.", field: "passing_year" };
  return { ok: true, value };
}

/** Lines for the enquiry and the staff email */
export function describeEducation(e: Education) {
  const scale = SCORE_SCALES.find((s) => s.id === e.scale);
  return [
    e.level && `Wants to study: ${levelLabel(e.level)}`,
    e.qualification && `Highest qualification: ${qualificationLabel(e.qualification)}${e.score ? `, ${e.score}${e.scale === "percent" ? "%" : ` (${scale?.label})`}` : ""}${e.passingYear ? `, finished ${e.passingYear}` : ""}`,
    e.test !== "none" && e.testScore ? `English: ${englishLabel(e.test)} ${e.testScore}` : e.qualification ? "English: no test yet" : "",
    e.backlogs ? `Backlogs: ${e.backlogs}` : "",
    e.city && `Lives in: ${e.city}`,
  ].filter(Boolean) as string[];
}

/** The same details in the student portal's own format */
export function educationForProfile(e: Education) {
  const out: Record<string, unknown> = {};
  if (e.level) out.preferred_level = PORTAL_LEVEL[e.level] ?? null;
  if (e.city) out.city = e.city;
  if (e.qualification)
    out.academic_history = [
      {
        level: PORTAL_QUALIFICATION[e.qualification] ?? "Other",
        institution: "",
        board: "",
        year: e.passingYear ? String(e.passingYear) : "",
        score: e.score ? (e.scale === "percent" ? `${e.score}%` : `${e.score} ${e.scale === "cgpa10" ? "CGPA (out of 10)" : "GPA (out of 4)"}`) : "",
      },
    ];
  if (e.test !== "none" && e.testScore) out.test_scores = [{ test: PORTAL_TEST[e.test] ?? "Other", score: String(e.testScore), date: "" }];
  return out;
}

/** 4560000 → "₹45.6 lakh" */
export function lakh(rupees: number) {
  const l = rupees / 1_00_000;
  if (l >= 100) return `₹${(l / 100).toFixed(1).replace(/\.0$/, "")} crore`;
  return `₹${l >= 10 ? Math.round(l) : l.toFixed(1).replace(/\.0$/, "")} lakh`;
}

function query(params: Record<string, string | undefined>) {
  const p = new URLSearchParams();
  for (const [k, v] of Object.entries(params)) if (v) p.set(k, k === "course" ? v.slice(0, 200) : v);
  const s = p.toString();
  return s ? `?${s}` : "";
}

export function locationLabel(country: string, value: string) {
  if (country === "australia") return AU_STATES[value] ?? value;

/** First-year fee of a course: the whole fee for courses shorter than a year */
const firstYear = (c: Course) => (!c.fee ? 0 : c.months >= 12 ? (c.fee * 12) / c.months : c.fee);

const median = (xs: number[]) => {
  if (!xs.length) return 0;
  const s = [...xs].sort((a, b) => a - b);
  const m = s.length >> 1;
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2;
};

function loadStore() {
  store ??= (async () => {
    const raw = await readJson<{ generatedAt: string; inr?: Record<string, number | string>; items: RawUni[] }>("universities.json");
    const list: Indexed<University>[] = [];
    const byId = new Map<string, University>();
    for (const u of raw?.items ?? []) {
      if (!isCountry(u.c) || !u.n) continue;
      const region = u.r ?? "";
      const city = u.t ?? "";
      const item: University = {
        id: u.i,
        name: u.n,
        country: u.c,
        region,
        city,
        website: u.w ?? "",
        size: u.k ?? 0,
        students: u.c === "usa" || u.c === "europe" || u.c === "new-zealand" ? (u.k ?? 0) : 0,
        currency: u.cur || COUNTRY_CURRENCY[u.c] || "",
        pgwp: !!u.p,
        type: u.ty === "public" || u.ty === "private" ? u.ty : u.c === "canada" ? (u.pub ? "public" : "private") : "",
        location: locationOf(u.c, region, city),
        courses: 0,
        feeYear: u.f ?? 0,
      };
      list.push({ item, hay: fold(`${item.name} ${item.city} ${item.region} ${locationLabel(item.country, item.region)}`) });
      byId.set(item.id, item);
    }
    const inr: Record<string, number> = { ...INR_FALLBACK };
    for (const [k, v] of Object.entries(raw?.inr ?? {})) if (typeof v === "number" && v > 0) inr[k] = v;

    // Course counts, levels, subjects and typical fees per university, from the course lists
    const stats = new Map<string, Stats>();
    const fees = new Map<string, Map<string, number[]>>();
    await Promise.all(
      CATALOGUE_COUNTRIES.filter((c) => c.courses).map(async (c) => {
        for (const { item: course } of await loadCourses(c.id, byId)) {
          const u = course.university;
          u.courses += 1;
          let st = stats.get(u.id);
          if (!st) stats.set(u.id, (st = { combos: new Map(), feeByLevel: new Map() }));
          const key = `${course.level}|${course.subject}`;
          st.combos.set(key, (st.combos.get(key) ?? 0) + 1);
          const fy = firstYear(course);
          if (fy) {
            let byLevel = fees.get(u.id);
            if (!byLevel) fees.set(u.id, (byLevel = new Map()));
            byLevel.set(course.level, [...(byLevel.get(course.level) ?? []), fy]);
            byLevel.set("*", [...(byLevel.get("*") ?? []), fy]);
          }
        }
      }),
    );
    for (const [id, byLevel] of fees) {
      const st = stats.get(id)!;
      for (const [level, xs] of byLevel) st.feeByLevel.set(level, Math.round(median(xs)));
      const u = byId.get(id)!;
      if (!u.feeYear) u.feeYear = st.feeByLevel.get("*") ?? 0;
    }
    return { generatedAt: raw?.generatedAt ?? null, list, byId, stats, inr };
  })();
  return store;
}

function loadCourses(country: string, byId: Map<string, University>) {
  let p = courseStores.get(country);
  if (!p) {
    p = (async () => {
      const raw = await readJson<{ items: RawCourse[] }>(`courses-${country}.json`);
      const out: Indexed<Course>[] = [];
      for (const [uid, title, level, subject, months, fee, url] of raw?.items ?? []) {
        const university = byId.get(uid);
        if (!university || !title) continue;
        out.push({ item: { university, title, level, subject, months: months || 0, fee: fee || 0, url: url || "" }, hay: fold(`${title} ${university.name} ${university.city} ${university.region}`) });
      }
      return out;
    })();
    courseStores.set(country, p);
  }
  return p;
}

const tokens = (q: string) => fold(q).split(" ").filter(Boolean).slice(0, 8);

/** `startBonus`: rank titles that start with the search first (courses); for universities a name
 * match is enough, and size decides, so "University of Toronto" isn't buried under "Toronto …" colleges. */
function score(hay: string, name: string, words: string[], phrase: string, startBonus = true) {
  if (!words.length) return 1;
  // whole words count more than parts of words
  let s = 0;
  for (const w of words) {
    const at = hay.indexOf(w);
    if (at < 0) return 0;
    s += at === 0 || hay[at - 1] === " " ? 2 : 1;
  }
  if (phrase.length > 2) {
    if (startBonus && name.startsWith(phrase)) s += 6;
    else if (name.includes(phrase)) s += 3;
  }
  return s;
}

export type UniversitySearch = { q?: string; country?: string; limit?: number; offset?: number };

export async function searchUniversities({ q = "", country, limit = 20, offset = 0 }: UniversitySearch) {
  const { list } = await loadStore();
  const words = tokens(q);
  const phrase = words.join(" ");
  const hits: { u: University; s: number }[] = [];
  for (const { item, hay } of list) {
    if (country && item.country !== country) continue;
    const s = score(hay, fold(item.name), words, phrase, false);
    if (s) hits.push({ u: item, s });
  }
  hits.sort((a, b) => b.s - a.s || b.u.size - a.u.size || a.u.name.localeCompare(b.u.name));
  return { total: hits.length, items: hits.slice(offset, offset + limit).map((h) => h.u) };
}

export type CourseSearch = { q?: string; country?: string; level?: string; subject?: string; university?: string; limit?: number; offset?: number };

export async function searchCourses({ q = "", country, level, subject, university, limit = 20, offset = 0 }: CourseSearch) {
  const { byId } = await loadStore();
  const countries = CATALOGUE_COUNTRIES.filter((c) => c.courses && (!country || c.id === country));
  const words = tokens(q);
  const phrase = words.join(" ");
  const hits: { c: Course; s: number }[] = [];
  for (const ct of countries) {
    for (const { item, hay } of await loadCourses(ct.id, byId)) {
      if (level && item.level !== level) continue;
      if (subject && item.subject !== subject) continue;
      if (university && item.university.id !== university) continue;
      const s = score(hay, fold(item.title), words, phrase);
      if (s) hits.push({ c: item, s });
    }
  }
  hits.sort(
    (a, b) =>
      b.s - a.s ||
      b.c.university.size - a.c.university.size ||
      a.c.university.name.localeCompare(b.c.university.name) ||
      a.c.title.localeCompare(b.c.title),
  );
  return { total: hits.length, items: hits.slice(offset, offset + limit).map((h) => h.c) };
}

/** Number of universities and courses per country, for the course finder's country picker. */
export async function catalogueCounts() {
  const { list, byId } = await loadStore();
  const out: Record<string, { universities: number; courses: number }> = {};
  for (const c of CATALOGUE_COUNTRIES) out[c.id] = { universities: 0, courses: 0 };
  for (const { item } of list) out[item.country].universities += 1;
  for (const c of CATALOGUE_COUNTRIES.filter((x) => x.courses)) out[c.id].courses = (await loadCourses(c.id, byId)).length;
  return out;
}
