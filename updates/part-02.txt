/**
 * All site content. Comes from the Content area (Sanity) once "Website settings"
 * exists there; until then the built-in defaults keep the site complete.
 */
export async function getSiteContent(): Promise<SiteContent> {
  const data = await sanityQuery<QueryResult>(SITE_QUERY);
  if (!data?.settings) return defaults();

  const fallback = defaults();
  const s = data.settings;
  return {
    fromCms: true,
    companyName: s.companyName || fallback.companyName,
    legalName: s.legalName || s.companyName || fallback.legalName,
    tagline: s.tagline || fallback.tagline,
    description: s.description || fallback.description,
    whatsappNumber: s.whatsappNumber ? digits(s.whatsappNumber) : fallback.whatsappNumber,
    whatsappMessage: s.whatsappMessage || fallback.whatsappMessage,
    contacts: s.contacts ?? [],
    socialLinks: (s.socialLinks ?? []).filter((l) => l.url),
    hero: {
      eyebrow: s.heroEyebrow ?? fallback.hero.eyebrow,
      titleLine1: s.heroTitleLine1 || fallback.hero.titleLine1,
      titleLine2: s.heroTitleLine2 ?? fallback.hero.titleLine2,
      intro: s.heroIntro ?? fallback.hero.intro,
      note: s.heroNote ?? fallback.hero.note,
    },
    stats: s.stats?.length ? s.stats : defaultStats((data.destinations ?? []).length || fallback.destinations.length, (data.services ?? []).length || fallback.services.length),
    intro: introSettings(s.intro),
    sections: { ...defaultSections, ...(s.sections ?? {}) },
    offices: (data.offices ?? []).map((o) => ({
      ...o,
      id: o._id,
      mapQuery: o.mapQuery || [o.legalName, ...o.addressLines].filter(Boolean).join(", "),
      hours: o.hours ?? [],
      contacts: o.contacts ?? [],
      photos: (o.photos ?? []).filter((p) => p?.asset?.url),
    })),
    ticker: [
      ...(data.tickerPosts ?? []).map((p) => ({ tag: p.kind === "event" ? "Event" : "News", text: p.title, link: `/news/${p.slug}` })),
      ...(data.ticker ?? []),
    ],
    banners: (data.banners ?? []).map((b) => ({ ...hasImage(b), id: b._id })),
    destinations: (data.destinations ?? [])
      .filter((d) => d.id)
      .map((d) => ({ ...d, countries: d.countries?.length ? d.countries : (destinationDetails[d.id]?.countries ?? null) })),
    services: (data.services ?? []).map((sv) => ({ ...sv, slug: sv.slug || slugify(sv.title) })),
    openNow: data.openNow ?? [],
    onshore: data.onshore ?? [],
    team: (data.team ?? []).map(hasPhoto),
    stories: (data.stories ?? []).map(hasPhoto),
    posts: (data.posts ?? []).map(hasImage),
    nextEvent: data.nextEvent ? hasImage(data.nextEvent) : null,
  };
}

const orList = <T,>(v: T[] | null | undefined, fb: T[]) => (v && v.length ? v : fb);

function fullDestination(d: Partial<Destination> & DestinationCard): Destination {
  const fb = destinationDetails[d.id];
  return {
    ...d,
    intro: d.intro || fb?.intro || d.summary,
    highlights: orList(d.highlights, fb?.highlights ?? []),
    popularCourses: orList(d.popularCourses, fb?.popularCourses ?? []),
    countries: orList(d.countries ?? null, fb?.countries ?? []),
    visaSteps: orList(d.visaSteps, fb?.visaSteps ?? []),
    postStudyWork: d.postStudyWork || fb?.postStudyWork || "",
    officialLinks: d.officialLinks ?? [],
    faqs: orList(d.faqs, destinationFaqs(d.name === "United Kingdom" ? "the UK" : d.name === "United States" ? "the USA" : d.name)),
    heroImage: d.heroImage?.asset?.url ? d.heroImage : null,
  };
}

function fullService(sv: Partial<ServiceFull> & { title: string; description: string }): ServiceFull {
  const slug = sv.slug || slugify(sv.title);
  const fb = serviceDetails.find((x) => x.slug === slug || x.title === sv.title);
  return {
    title: sv.title,
    slug,
    description: sv.description,
    icon: sv.icon ?? fb?.icon,
    steps: orList(sv.steps, fb?.steps ?? []),
    included: orList(sv.included, fb?.included ?? []),
  };
}

export async function getLibrary(): Promise<Library> {
  const data = await sanityQuery<Partial<Library> & { destinations?: (Partial<Destination> & DestinationCard)[] }>(LIBRARY_QUERY);
  const site = await getSiteContent();
  const destinations = data?.destinations?.length
    ? data.destinations.filter((d) => d.id).map(fullDestination)
    : site.destinations.map((d) => fullDestination(d));
  const services = data?.services?.length ? data.services.map(fullService) : serviceDetails.map(fullService);
  return {
    destinations,
    services,
    stories: (data?.stories ?? []).map(hasPhoto),
    posts: (data?.posts ?? []).map(hasImage),
    faqs: data?.faqs?.length ? data.faqs : defaultFaqs,
    universities: data?.universities ?? [],
    scholarships: data?.scholarships ?? [],
  };
}

/**
 * Logo intro settings (Content → Website settings → Logo intro).
 * Timeline for a total of T seconds:
 *   0 → about 1.1 s × pace   logo draws, name and tagline rise
 *   … hold                   finished logo stays on screen
 *   T − 0.7 s                name and tagline fade out (0.2 s), background still solid
 *   T − 0.5 s                background fades and the logo glides into the header (0.5 s)
 *   T                        intro removed; the page has already started to appear
 */
export type IntroSettings = { show: boolean; seconds: number; replayMinutes: number };

export const defaultIntro: IntroSettings = { show: true, seconds: 2.5, replayMinutes: 30 };

export function introSettings(raw?: Partial<IntroSettings> | null): IntroSettings {
  const seconds = Number(raw?.seconds);
  const replay = Number(raw?.replayMinutes);
  return {
    show: raw?.show !== false,
    seconds: Number.isFinite(seconds) && seconds > 0 ? Math.min(5, Math.max(1.8, seconds)) : defaultIntro.seconds,
    replayMinutes: Number.isFinite(replay) && replay >= 0 ? Math.min(1440, replay) : defaultIntro.replayMinutes,
  };
}

/** CSS custom properties set on <html>: total length and drawing pace. */
export function introStyle(s: IntroSettings): Record<string, string> {
  const pace = Math.min(1.5, Math.max(1, s.seconds / 2));
  return { "--intro-total": `${s.seconds}s`, "--intro-pace": pace.toFixed(3) };
}

/**
 * Runs in <head> before first paint: marks that JavaScript is on (for scroll
 * reveals) and decides whether the intro plays, so it never flashes.
 * ?intro=1 on any address always plays it.
 */
export function introScript(s: IntroSettings) {
  const replayMs = Math.round(s.replayMinutes * 60000);
  return `(function(){var d=document.documentElement;d.dataset.js="";try{if(location.search.indexOf("intro=1")>=0)return;${
    s.show ? "" : 'd.dataset.intro="skip";return;'
  }var t=+localStorage.getItem("es-intro")||0;if(${replayMs}>0&&Date.now()-t<${replayMs})d.dataset.intro="skip"}catch(e){}})()`;
}

type PlaceResponse = {
  displayName?: { text?: string };
  rating?: number;
  userRatingCount?: number;
  googleMapsUri?: string;
  reviews?: {
    rating?: number;
    text?: { text?: string };

    relativePublishTimeDescription?: string;
    googleMapsUri?: string;
    authorAttribution?: { displayName?: string; uri?: string; photoUri?: string };

const API = "https://places.googleapis.com/v1";

/** Finds a Place ID from the office name and address. "IDs only" text search has no charge. */
async function findPlaceId(key: string, query: string): Promise<string | null> {
  const res = await fetch(`${API}/places:searchText`, {
    method: "POST",
    headers: { "Content-Type": "application/json", "X-Goog-Api-Key": key, "X-Goog-FieldMask": "places.id" },
    body: JSON.stringify({ textQuery: query, pageSize: 1 }),
  });
  if (!res.ok) throw new Error(`Text search ${res.status}`);
  const data = (await res.json()) as { places?: { id: string }[] };
  return data.places?.[0]?.id ?? null;
}

async function loadReviews(placeId: string | null, query: string): Promise<PlaceReviews | null> {
  "use cache";
  cacheTag("google-reviews");
  // Refreshed about twice a day: well inside Google's free monthly allowance.
  cacheLife({ stale: 3600, revalidate: 43200, expire: 86400 });

  const key = process.env.GOOGLE_PLACES_API_KEY;
  if (!key) return null;
  try {
    const id = placeId || (await findPlaceId(key, query));
    if (!id) return null;
    const res = await fetch(`${API}/places/${encodeURIComponent(id)}?languageCode=en`, {
      headers: {
        "X-Goog-Api-Key": key,
        "X-Goog-FieldMask": "displayName,rating,userRatingCount,googleMapsUri,reviews",
      },
    });
    if (!res.ok) throw new Error(`Place details ${res.status}`);
    const p = (await res.json()) as PlaceResponse;
    return {
      name: p.displayName?.text ?? "",
      rating: p.rating ?? null,
      count: p.userRatingCount ?? null,
      mapsUrl: p.googleMapsUri ?? null,
      reviews: (p.reviews ?? [])
        .map((r) => ({
          author: r.authorAttribution?.displayName ?? "Google user",
          authorUrl: r.authorAttribution?.uri ?? null,
          authorPhoto: r.authorAttribution?.photoUri ?? null,
          rating: r.rating ?? 0,
          text: r.text?.text ?? r.originalText?.text ?? "",
          when: r.relativePublishTimeDescription ?? "",
          url: r.googleMapsUri ?? null,
        }))
        .filter((r) => r.text),
    };
  } catch (error) {
    console.error("[reviews] Google Places request failed:", error instanceof Error ? error.message : error);
    return null;
  }
}

/**
 * Live Google rating and reviews for an office (Google returns up to five).
 * Returns null, and the reviews block stays hidden, until GOOGLE_PLACES_API_KEY
 * is set in Netlify. Reviews are shown exactly as Google returns them.
 */
export async function getPlaceReviews(office: { googlePlaceId?: string | null; mapQuery: string } | undefined) {
  if (!office || !process.env.GOOGLE_PLACES_API_KEY) return null;
  return loadReviews(office.googlePlaceId ?? null, office.mapQuery);
}
