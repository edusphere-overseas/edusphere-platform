"use client";

import Link from "next/link";
import { useActionState, useState } from "react";
import { Eye, EyeOff, GraduationCap, MailCheck, Users } from "lucide-react";
import {
  completeOnboarding,
  requestPasswordReset,
  resendVerification,
  setNewPassword,
  signIn,
  signInWithGoogle,
  signUp,
  type FormState,
} from "@/app/(auth)/actions";
import { consentText } from "@/lib/portal/consent";
import { FormError, SubmitButton, Turnstile } from "./FormBits";
import { btnGhost, btnPrimary, hint, input, label } from "./ui";

const googleOn = process.env.NEXT_PUBLIC_GOOGLE_SIGN_IN === "on";

export function AuthCard({ title, intro, children }: { title: string; intro?: React.ReactNode; children: React.ReactNode }) {
  return (
    <div className="rounded-3xl bg-white p-6 text-ink shadow-[0_30px_80px_-30px_rgba(0,0,0,0.6)] sm:p-8">
      <h1 className="text-2xl font-extrabold tracking-[-0.02em] text-deep">{title}</h1>
      {intro && <p className="mt-2 text-[0.95rem] leading-relaxed text-slate">{intro}</p>}
      <div className="mt-6">{children}</div>
    </div>
  );
}

function GoogleButton({ next, role, text }: { next?: string; role?: string; text: string }) {
  if (!googleOn) return null;
  return (
    <form action={signInWithGoogle}>
      <input type="hidden" name="next" value={next ?? ""} />
      {role && <input type="hidden" name="role" value={role} />}
      <button type="submit" className={`${btnGhost} w-full`}>
        <svg viewBox="0 0 24 24" className="h-4.5 w-4.5" aria-hidden="true">
          <path fill="#4285F4" d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92a5.06 5.06 0 0 1-2.2 3.32v2.77h3.57c2.08-1.92 3.27-4.74 3.27-8.1z" />
          <path fill="#34A853" d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84A11 11 0 0 0 12 23z" />
          <path fill="#FBBC05" d="M5.84 14.1A6.6 6.6 0 0 1 5.5 12c0-.73.13-1.44.34-2.1V7.06H2.18A11 11 0 0 0 1 12c0 1.77.42 3.45 1.18 4.94l3.66-2.84z" />
          <path fill="#EA4335" d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1A11 11 0 0 0 2.18 7.06l3.66 2.84C6.71 7.31 9.14 5.38 12 5.38z" />
        </svg>
        {text}
      </button>
    </form>
  );
}

function Divider() {
  if (!googleOn) return null;
  return (
    <div className="my-5 flex items-center gap-3 text-xs font-semibold uppercase tracking-[0.14em] text-slate/70">
      <span className="h-px flex-1 bg-slate/20" />
      or
      <span className="h-px flex-1 bg-slate/20" />
    </div>
  );
}

function PasswordInput({ name, autoComplete, id }: { name: string; autoComplete: string; id: string }) {
  const [show, setShow] = useState(false);
  return (
    <div className="relative">
      <input id={id} name={name} type={show ? "text" : "password"} required minLength={8} autoComplete={autoComplete} className={`${input} pr-12`} />
      <button
        type="button"
        onClick={() => setShow((s) => !s)}
        className="absolute inset-y-0 right-0 grid w-12 place-items-center text-slate hover:text-deep"
        aria-label={show ? "Hide password" : "Show password"}
        aria-pressed={show}
      >
        {show ? <EyeOff className="h-4.5 w-4.5" /> : <Eye className="h-4.5 w-4.5" />}
      </button>
    </div>
  );
}

function CheckInbox({ email, kind }: { email: string; kind: "verify" | "reset" }) {
  const [state, resend] = useActionState(resendVerification, null);
  return (
    <div className="text-center" role="status">
      <div className="mx-auto grid h-14 w-14 place-items-center rounded-2xl bg-teal/12 text-teal">
        <MailCheck className="h-7 w-7" aria-hidden="true" />
      </div>
      <p className="mt-4 text-lg font-bold text-deep">Check your inbox</p>
      <p className="mt-2 leading-relaxed text-slate">
        {kind === "verify" ? (
          <>
            We&apos;ve sent a link to <strong className="text-deep">{email}</strong>. Open it to verify your email and finish creating your account.
          </>
        ) : (
          <>
            If <strong className="text-deep">{email}</strong> has an account, we&apos;ve sent a link to choose a new password. It works for one hour.
          </>
        )}
      </p>
      <p className="mt-3 text-sm text-slate">Nothing after a few minutes? Check your spam folder.</p>
      {kind === "verify" && (
        <form action={resend} className="mt-5">
          <input type="hidden" name="email" value={email} />
          <Turnstile resetKey={state} />
          <SubmitButton className={btnGhost} pending="Sending…">
            Send the link again
          </SubmitButton>
          {state?.done && <p className="mt-3 text-sm font-medium text-[#17663a]">Sent again.</p>}
          <FormError message={state?.error} />
        </form>
      )}
    </div>
  );
}

export function SignInForm({ next, error }: { next?: string; error?: string }) {
  const [state, action] = useActionState<FormState, FormData>(signIn, null);
  const linkError =
    error === "link"
      ? "That link has expired or was already used. Sign in, or ask for a new link."
      : error === "google"
        ? "Google sign-in didn't finish. Please try again."
        : undefined;

  return (
    <AuthCard title="Sign in" intro="Welcome back. Students and parents sign in here.">
      <GoogleButton next={next} text="Continue with Google" />
      <Divider />
      <form action={action} className="grid gap-4">
        <input type="hidden" name="next" value={next ?? ""} />
        <label className={label}>
          Email
          <input name="email" type="email" required autoComplete="email" defaultValue={state?.values?.email} className={input} />
        </label>
        <div className={label}>
          <span className="flex items-center justify-between">
            <label htmlFor="signin-password">Password</label>
            <Link href="/forgot-password" className="text-xs font-semibold text-teal hover:text-navy">
              Forgot password?
            </Link>
          </span>
          <PasswordInput id="signin-password" name="password" autoComplete="current-password" />
        </div>
        <Turnstile resetKey={state} />
        <FormError message={state?.error ?? linkError} />
        <SubmitButton className={`${btnPrimary} w-full`} pending="Signing in…">
          Sign in
        </SubmitButton>
      </form>
      {state?.unverifiedEmail && <ResendBox email={state.unverifiedEmail} />}
      <p className="mt-6 text-center text-sm text-slate">
        New here?{" "}
        <Link href={next ? `/sign-up?next=${encodeURIComponent(next)}` : "/sign-up"} className="font-semibold text-teal hover:text-navy">
          Create an account
        </Link>
      </p>
    </AuthCard>
  );
}

function ResendBox({ email }: { email: string }) {
  const [state, resend] = useActionState(resendVerification, null);
  return (
    <form action={resend} className="mt-4 rounded-2xl bg-mist p-4 text-sm">
      <input type="hidden" name="email" value={email} />
      <p className="text-slate">Didn&apos;t get the verification email?</p>
      <Turnstile resetKey={state} />
      <div className="mt-2">
        <SubmitButton className={btnGhost} pending="Sending…">
          Send it again
        </SubmitButton>
      </div>
      {state?.done && <p className="mt-2 font-medium text-[#17663a]">Sent. Check your inbox.</p>}
      <FormError message={state?.error} />
    </form>
  );
}

function RoleChoice({ role, setRole }: { role: string; setRole: (r: "student" | "parent") => void }) {
  const opts = [
    { value: "student" as const, title: "I'm a student", text: "Planning to study abroad", Icon: GraduationCap },
    { value: "parent" as const, title: "I'm a parent", text: "Or a legal guardian", Icon: Users },
  ];
  return (
    <fieldset>
      <legend className="sr-only">Account type</legend>
      <div className="grid grid-cols-2 gap-3">
        {opts.map(({ value, title, text, Icon }) => (
          <label
            key={value}
            className={`flex cursor-pointer flex-col gap-2 rounded-2xl p-4 ring-1 transition has-[:focus-visible]:ring-3 has-[:focus-visible]:ring-teal/40 ${
              role === value ? "bg-teal/8 ring-2 ring-teal" : "ring-slate/25 hover:bg-mist"
            }`}
          >
            <input type="radio" name="role-choice" value={value} checked={role === value} onChange={() => setRole(value)} className="sr-only" />
            <Icon className={`h-6 w-6 ${role === value ? "text-teal" : "text-slate"}`} aria-hidden="true" />
            <span className="text-sm font-bold text-deep">{title}</span>
            <span className="text-xs leading-snug text-slate">{text}</span>
          </label>
        ))}
      </div>
    </fieldset>
  );
}

export function SignUpForm({ next, initialRole }: { next?: string; initialRole?: string }) {
  const [state, action] = useActionState<FormState, FormData>(signUp, null);
  const [role, setRole] = useState<"student" | "parent">(
    (state?.values?.role as "student" | "parent") ?? (initialRole === "parent" ? "parent" : "student"),
  );

  if (state?.done) {
    return (
      <AuthCard title="Almost there">
        <CheckInbox email={state.done} kind="verify" />
      </AuthCard>
    );
  }

  return (
    <AuthCard title="Create your account" intro="Upload documents, follow every stage and request services, in one private place.">
      <div className="mb-4 grid gap-4">
        <RoleChoice role={role} setRole={setRole} />
        <GoogleButton role={role} next={next} text={`Sign up with Google as a ${role}`} />
      </div>
      <Divider />
      <form action={action} className="grid gap-4">
        <input type="hidden" name="role" value={role} />
        <label className={label}>
          Full name
          <input name="full_name" required autoComplete="name" maxLength={120} defaultValue={state?.values?.full_name} className={input} />
        </label>
        <label className={label}>
          Email
          <input name="email" type="email" required autoComplete="email" defaultValue={state?.values?.email} className={input} />
        </label>
        <label className={label}>
          <span>
            Mobile number <span className="font-normal text-slate">(optional)</span>
          </span>
          <input name="phone" type="tel" autoComplete="tel" inputMode="tel" maxLength={30} placeholder="+91 …" defaultValue={state?.values?.phone} className={input} />
        </label>
        {role === "student" && (
          <label className={label}>
            Date of birth
            <input name="date_of_birth" type="date" required max={new Date().toISOString().slice(0, 10)} defaultValue={state?.values?.date_of_birth} className={input} />
            <span className={hint}>If you&apos;re under 18, a parent or guardian links to your account before you upload documents.</span>
          </label>
        )}
        <div className={label}>
          <label htmlFor="signup-password">Password</label>
          <PasswordInput id="signup-password" name="password" autoComplete="new-password" />
          <span className={hint}>At least 8 characters, with letters and numbers.</span>
        </div>
        <label className="flex items-start gap-3 text-sm leading-relaxed text-slate">
          <input name="consent" type="checkbox" required className="mt-0.5 h-5 w-5 shrink-0 accent-teal" />
          <span>{role === "student" ? consentText.student : consentText.parent}</span>
        </label>
        <p className="-mt-2 pl-8 text-xs text-slate">
          How we use your data:{" "}
          <Link href="/privacy" target="_blank" className="font-semibold text-teal hover:text-navy">
            Privacy policy
          </Link>
        </p>
        <Turnstile resetKey={state} />
        <FormError message={state?.error} />
        <SubmitButton className={`${btnPrimary} w-full`} pending="Creating your account…">
          Create account
        </SubmitButton>
      </form>
      <p className="mt-6 text-center text-sm text-slate">
        Already have an account?{" "}
        <Link href="/sign-in" className="font-semibold text-teal hover:text-navy">
          Sign in
        </Link>
      </p>
    </AuthCard>
  );
}

export function ForgotPasswordForm() {
  const [state, action] = useActionState<FormState, FormData>(requestPasswordReset, null);
  if (state?.done) {
    return (
      <AuthCard title="Reset your password">
        <CheckInbox email={state.done} kind="reset" />
      </AuthCard>
    );
  }
  return (
    <AuthCard title="Reset your password" intro="Enter your email and we'll send you a link to choose a new password.">
      <form action={action} className="grid gap-4">
        <label className={label}>
          Email
          <input name="email" type="email" required autoComplete="email" defaultValue={state?.values?.email} className={input} />
        </label>
        <Turnstile resetKey={state} />
        <FormError message={state?.error} />
        <SubmitButton className={`${btnPrimary} w-full`} pending="Sending…">
          Send the link
        </SubmitButton>
      </form>
      <p className="mt-6 text-center text-sm text-slate">
        <Link href="/sign-in" className="font-semibold text-teal hover:text-navy">
          Back to sign in
        </Link>
      </p>
    </AuthCard>
  );
}

export function NewPasswordForm({ welcome }: { welcome?: boolean }) {
  const [state, action] = useActionState<FormState, FormData>(setNewPassword, null);
  return (
    <AuthCard
      title={welcome ? "Welcome to Edusphere" : "Choose a new password"}
      intro={welcome ? "Your parent created your account. Choose a password to finish setting it up." : undefined}
    >
      <form action={action} className="grid gap-4">
        <div className={label}>
          <label htmlFor="new-password">New password</label>
          <PasswordInput id="new-password" name="password" autoComplete="new-password" />

        </div>
        <div className={label}>
          <label htmlFor="confirm-password">Type it again</label>
          <PasswordInput id="confirm-password" name="confirm" autoComplete="new-password" />
        </div>
        <FormError message={state?.error} />
        <SubmitButton className={`${btnPrimary} w-full`} pending="Saving…">
          Save password
        </SubmitButton>
      </form>
    </AuthCard>
  );
}

export function WelcomeForm({
  role: fixedRole,
  suggestedRole,
  name,
  phone,
  dob,
  invited,
}: {
  role: string | null;
  suggestedRole?: string;
  name: string;
  phone: string;
  dob: string;
  invited: boolean;
}) {
  const [state, action] = useActionState<FormState, FormData>(completeOnboarding, null);
  const [role, setRole] = useState<"student" | "parent">(
    (fixedRole as "student" | "parent") ?? (state?.values?.role as "student" | "parent") ?? (suggestedRole === "parent" ? "parent" : "student"),
  );
  return (
    <AuthCard
      title="One more step"
      intro={invited ? "Check your details and give your consent to start using the portal." : "Tell us who you are, so we show you the right things."}
    >
      <form action={action} className="grid gap-4">
        <input type="hidden" name="role" value={fixedRole ?? role} />
        {!fixedRole && <RoleChoice role={role} setRole={setRole} />}
        <label className={label}>
          Full name
          <input name="full_name" required autoComplete="name" maxLength={120} defaultValue={state?.values?.full_name ?? name} className={input} />
        </label>
        <label className={label}>
          <span>
            Mobile number <span className="font-normal text-slate">(optional)</span>
          </span>
          <input name="phone" type="tel" autoComplete="tel" inputMode="tel" maxLength={30} placeholder="+91 …" defaultValue={state?.values?.phone ?? phone} className={input} />
        </label>
        {role === "student" && (
          <label className={label}>
            Date of birth
            <input
              name="date_of_birth"
              type="date"
              required
              max={new Date().toISOString().slice(0, 10)}
              defaultValue={state?.values?.date_of_birth ?? dob}
              readOnly={!!dob}
              className={input}
            />
            <span className={hint}>If you&apos;re under 18, a parent or guardian links to your account before you upload documents.</span>
          </label>
        )}
        <label className="flex items-start gap-3 text-sm leading-relaxed text-slate">
          <input name="consent" type="checkbox" required className="mt-0.5 h-5 w-5 shrink-0 accent-teal" />
          <span>{role === "student" ? consentText.student : consentText.parent}</span>
        </label>
        <p className="-mt-2 pl-8 text-xs text-slate">
          How we use your data:{" "}
          <Link href="/privacy" target="_blank" className="font-semibold text-teal hover:text-navy">
            Privacy policy
          </Link>
        </p>
        <FormError message={state?.error} />
        <SubmitButton className={`${btnPrimary} w-full`} pending="Saving…">
          Continue to the portal
        </SubmitButton>
      </form>
    </AuthCard>
  );
}
