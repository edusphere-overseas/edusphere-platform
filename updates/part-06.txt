type Phase = { step: "idle" } | { step: "sending"; pct: number } | { step: "checking" } | { step: "done" } | { step: "error"; message: string };

const SHRINK_OVER = 3 * 1024 * 1024; // photos larger than this are resized before sending
const MAX_SIDE = 2400;

/** Large phone photos become a sharp JPEG of at most 2400 px: quicker on mobile data, still easy to read. */
async function shrinkPhoto(file: File): Promise<File> {
  try {
    const bitmap = await createImageBitmap(file);
    const scale = Math.min(1, MAX_SIDE / Math.max(bitmap.width, bitmap.height));
    const canvas = document.createElement("canvas");
    canvas.width = Math.round(bitmap.width * scale);
    canvas.height = Math.round(bitmap.height * scale);
    canvas.getContext("2d")!.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
    const blob = await new Promise<Blob | null>((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.85));
    if (!blob || blob.size >= file.size) return file;
    return new File([blob], file.name.replace(/\.[^.]+$/, "") + ".jpg", { type: "image/jpeg" });
  } catch {
    return file;
  }
}

function send(url: string, file: File, onProgress: (pct: number) => void) {
  return new Promise<void>((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open("PUT", url);
    xhr.setRequestHeader("Content-Type", file.type);
    xhr.upload.onprogress = (e) => e.lengthComputable && onProgress(Math.round((e.loaded / e.total) * 100));
    xhr.onload = () => (xhr.status >= 200 && xhr.status < 300 ? resolve() : reject(new Error(`status ${xhr.status}`)));
    xhr.onerror = () => reject(new Error("network"));
    xhr.send(file);
  });
}

export function Uploader({ studentId, docType, replaces, serviceRequestId, askLabel, text = "Upload", primary = true }: Props) {
  const [phase, setPhase] = useState<Phase>({ step: "idle" });
  const [label, setLabel] = useState("");
  const fileInput = useRef<HTMLInputElement>(null);
  const cameraInput = useRef<HTMLInputElement>(null);
  const id = useId();
  const busy = phase.step === "sending" || phase.step === "checking";

  async function handle(original: File | undefined) {
    if (!original) return;
    if (askLabel && !label.trim()) {
      setPhase({ step: "error", message: "Please say what this document is first." });
      return;
    }
    let file = original;
    if (!(ALLOWED_TYPES as readonly string[]).includes(file.type)) {
      setPhase({ step: "error", message: "Please choose a PDF, JPG or PNG file." });
      return;
    }
    if (file.type !== "application/pdf" && file.size > SHRINK_OVER) file = await shrinkPhoto(file);
    if (file.size > MAX_FILE_BYTES) {
      setPhase({ step: "error", message: "That file is over 10 MB. Please choose a smaller file or scan at a lower resolution." });
      return;
    }

    setPhase({ step: "sending", pct: 0 });
    const start = await startUpload({ studentId, docType, label, fileName: file.name, mime: file.type, size: file.size, serviceRequestId });
    if ("error" in start) {
      setPhase({ step: "error", message: start.error });
      return;
    }
    try {
      await send(start.url, file, (pct) => setPhase({ step: "sending", pct }));
    } catch {
      await cancelUpload(start.id);
      setPhase({ step: "error", message: "The upload didn't finish. Check your connection and try again." });
      return;
    }
    setPhase({ step: "checking" });
    const done = await finishUpload(start.id, replaces);
    if ("error" in done) {
      setPhase({ step: "error", message: done.error });
      return;
    }
    setLabel("");
    setPhase({ step: "done" });
    setTimeout(() => setPhase((p) => (p.step === "done" ? { step: "idle" } : p)), 3000);
  }

  return (
    <div className="grid gap-2">
      {askLabel && (
        <label className="grid gap-1 text-xs font-semibold text-slate">
          What is this document?
          <input value={label} onChange={(e) => setLabel(e.target.value)} maxLength={120} placeholder="For example Gap year letter" className={`${input} py-2.5`} />
        </label>
      )}
      <div className="flex flex-wrap items-center gap-2">
        <input
          ref={fileInput}
          id={`${id}-file`}
          type="file"
          accept=".pdf,.jpg,.jpeg,.png,application/pdf,image/jpeg,image/png"
          className="sr-only"
          tabIndex={-1}
          onChange={(e) => {
            void handle(e.target.files?.[0]);
            e.target.value = "";
          }}
        />
        <input
          ref={cameraInput}
          type="file"
          accept="image/jpeg,image/png"
          capture="environment"
          className="sr-only"
          tabIndex={-1}
          onChange={(e) => {
            void handle(e.target.files?.[0]);
            e.target.value = "";
          }}
        />
        <button type="button" disabled={busy} onClick={() => fileInput.current?.click()} className={`${primary ? btnPrimary : btnGhost} min-h-10 py-2`}>
          {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" /> : <Upload className="h-4 w-4" aria-hidden="true" />}
          {phase.step === "sending" ? `Sending ${phase.pct}%` : phase.step === "checking" ? "Checking…" : text}

    banners: defaultBanners.map((b, i) => ({ id: `default-${i}`, ...b })),
    destinations: defaultDestinations.map((d) => ({
      id: d.id,
      name: d.name,
      flag: d.flag,
      summary: d.line,
      intakes: d.intakes,
      countries: destinationDetails[d.id]?.countries ?? null,
    })),
    services: serviceDetails.map(({ title, slug, description, icon }) => ({ title, slug, description, icon })),
    openNow: [],
    onshore: [],
    team: [],
    stories: [],
    posts: [],
    nextEvent: null,
  };
}

type Raw<T> = T & { [k: string]: unknown };
type QueryResult = {
  settings: Raw<{
    companyName?: string;
    legalName?: string;
    tagline?: string;
    description?: string;
    whatsappNumber?: string;
    whatsappMessage?: string;
    contacts?: ContactPoint[] | null;
    socialLinks?: SocialLink[] | null;
    heroEyebrow?: string;
    heroTitleLine1?: string;
    heroTitleLine2?: string;
    heroIntro?: string;
    heroNote?: string;
    stats?: { value: string; label: string }[] | null;
    intro?: Partial<IntroSettings> | null;
    sections?: Partial<Sections> | null;
    portalServices?: Partial<Record<keyof PortalServiceSettings, Partial<PortalServiceSetting> | null>> | null;
  }> | null;
  offices: (Omit<Office, "id" | "contacts" | "photos" | "mapQuery" | "hours"> & {
    _id: string;
    mapQuery?: string | null;
    hours?: string[] | null;
    contacts?: ContactPoint[] | null;
    photos?: SanityPhoto[] | null;
  })[];
  ticker: SiteContent["ticker"];
  tickerPosts?: { kind?: string | null; title: string; slug: string }[] | null;
  banners: (Omit<Banner, "id"> & { _id: string })[];
  destinations: DestinationCard[];
  services: (Omit<Service, "slug"> & { slug?: string | null })[];
  openNow: OpenNowItem[];
  onshore: SiteContent["onshore"];
  team: SiteContent["team"];
  stories: Story[] | null;
  posts: PostCard[] | null;
  nextEvent: PostCard | null;
};

/**
 * All site content. Comes from the Content area (Sanity) once "Website settings"
 * exists there; until then the built-in defaults keep the site complete.
 */
export async function getSiteContent(): Promise<SiteContent> {
  const data = await sanityQuery<QueryResult>(SITE_QUERY);
  if (!data?.settings) return defaults();

  const fallback = defaults();
  const s = data.settings;
  return {
    fromCms: true,
    companyName: s.companyName || fallback.companyName,
    legalName: s.legalName || s.companyName || fallback.legalName,
    tagline: s.tagline || fallback.tagline,
    description: s.description || fallback.description,
    whatsappNumber: s.whatsappNumber ? digits(s.whatsappNumber) : fallback.whatsappNumber,
    whatsappMessage: s.whatsappMessage || fallback.whatsappMessage,
    contacts: s.contacts ?? [],
    socialLinks: (s.socialLinks ?? []).filter((l) => l.url),
    hero: {
      eyebrow: s.heroEyebrow ?? fallback.hero.eyebrow,
      titleLine1: s.heroTitleLine1 || fallback.hero.titleLine1,
      titleLine2: s.heroTitleLine2 ?? fallback.hero.titleLine2,
      intro: s.heroIntro ?? fallback.hero.intro,
      note: s.heroNote ?? fallback.hero.note,
    },
    stats: s.stats?.length ? s.stats : defaultStats((data.destinations ?? []).length || fallback.destinations.length, (data.services ?? []).length || fallback.services.length),
    intro: introSettings(s.intro),
    sections: { ...defaultSections, ...(s.sections ?? {}) },
    portalServices: portalServiceSettings(s.portalServices),
    offices: (data.offices ?? []).map((o) => ({
      ...o,
      id: o._id,
      mapQuery: o.mapQuery || [o.legalName, ...o.addressLines].filter(Boolean).join(", "),
      hours: o.hours ?? [],
      contacts: o.contacts ?? [],
      photos: (o.photos ?? []).filter((p) => p?.asset?.url),
    })),
    ticker: [
      ...(data.tickerPosts ?? []).map((p) => ({ tag: p.kind === "event" ? "Event" : "News", text: p.title, link: `/news/${p.slug}` })),
      ...(data.ticker ?? []),
    ],
    banners: (data.banners ?? []).map((b) => ({ ...hasImage(b), id: b._id })),
    destinations: (data.destinations ?? [])
      .filter((d) => d.id)
      .map((d) => ({ ...d, countries: d.countries?.length ? d.countries : (destinationDetails[d.id]?.countries ?? null) })),
    services: (data.services ?? []).map((sv) => ({ ...sv, slug: sv.slug || slugify(sv.title) })),
    openNow: data.openNow ?? [],
    onshore: data.onshore ?? [],
    team: (data.team ?? []).map(hasPhoto),
    stories: (data.stories ?? []).map(hasPhoto),
    posts: (data.posts ?? []).map(hasImage),
    nextEvent: data.nextEvent ? hasImage(data.nextEvent) : null,
  };
}
