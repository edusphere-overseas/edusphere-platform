const singletonActions = new Set(["publish", "discardChanges", "restore"]);

/**
 * After Publish, ask the website to show the change straight away (it otherwise refreshes
 * on its own within a minute or two). The publish takes a moment to land, so try a few times.
 */
async function refreshWebsite() {
  for (const wait of [1500, 2500, 4000, 8000]) {
    await new Promise((resolve) => setTimeout(resolve, wait));
    try {
      const res = await fetch("/api/revalidate", { method: "POST" });
      if (res.ok && (await res.json())?.revalidated) return true;
    } catch {
      // try again
    }
  }
  return false;
}

/** The normal Publish button, plus a website refresh and a note saying when it's live. */
function withWebsiteRefresh(Publish: DocumentActionComponent): DocumentActionComponent {
  const PublishAndRefresh: DocumentActionComponent = (props) => {
    const toast = useToast();
    const action = Publish(props);
    if (!action) return action;
    return {
      ...action,
      onHandle: () => {
        action.onHandle?.();
        void refreshWebsite().then((ok) =>
          toast.push({
            status: ok ? "success" : "info",
            title: ok ? "Live on the website" : "Published",
            description: ok ? "Reload the website page to see your change." : "The website will show it within a few minutes.",
          }),
        );
      },
    };
  };
  PublishAndRefresh.action = Publish.action;
  return PublishAndRefresh;
}

function StudioIcon() {
  return <LogoMark className="h-full w-full" />;
}

export default defineConfig({
  name: "edusphere",
  title: "Edusphere Overseas",
  icon: StudioIcon,
  projectId,
  dataset,
  basePath: studioBasePath,
  apiVersion,
  schema: {
    types: schemaTypes,
    // No "create new" for one-of-a-kind documents such as Website settings
    templates: (templates) => templates.filter(({ schemaType }) => !singletonTypes.has(schemaType)),
  },
  document: {
    actions: (input, context) =>
      (singletonTypes.has(context.schemaType) ? input.filter(({ action }) => action && singletonActions.has(action)) : input).map(
        (action) => (action.action === "publish" ? withWebsiteRefresh(action) : action),
      ),
  },
  plugins: [structureTool({ structure, title: "Content" })],
  tools: (prev) => [
    ...prev,
    { name: "starter-content", title: "Starter content", icon: PackagePlus, component: StarterContent },
  ],
});

import { revalidateTag } from "next/cache";
import type { NextRequest } from "next/server";
import { parseBody } from "next-sanity/webhook";
import { client } from "@/sanity/client";
import { SANITY_TAG } from "@/sanity/tags";

/** How recent a publish must be for the content area's refresh request to count. */
const RECENT_MS = 3 * 60 * 1000;

/** Latest publish already refreshed by this server instance, so repeat calls do nothing. */
let refreshedFor = "";

/**
 * Makes the website show newly published content straight away, instead of within the
 * usual minute or two. Two callers:
 *
 * 1. The content area (Sanity Studio at /admin/content) after each Publish. This needs no
 *    secret: the server asks Sanity itself whether something was published in the last few
 *    minutes and only then refreshes, so calling it at other times does nothing.
 * 2. Optionally a Sanity webhook, signed with SANITY_REVALIDATE_SECRET (set in Vercel and on
 *    the webhook). This also covers deletes and changes made outside the content area.
 */
export async function POST(req: NextRequest) {
  const secret = process.env.SANITY_REVALIDATE_SECRET;
  if (secret && req.headers.has("sanity-webhook-signature")) {
    const { isValidSignature, body } = await parseBody<{ _type?: string }>(req, secret);
    if (!isValidSignature) return new Response("Invalid signature", { status: 401 });
    revalidateTag(SANITY_TAG, { expire: 0 });
    return Response.json({ revalidated: true, type: body?._type ?? null });
  }

  let latest: string | null = null;
  try {
    latest = await client.fetch<string | null>(
      `*[!(_id in path("drafts.**")) && !(_id in path("versions.**"))] | order(_updatedAt desc)[0]._updatedAt`,
      {},
      { cache: "no-store" },
    );
  } catch (error) {
    console.error("[revalidate] couldn't reach Sanity:", error instanceof Error ? error.message : error);
    return Response.json({ revalidated: false }, { status: 502 });
  }

  const publishedAt = latest ? Date.parse(latest) : NaN;
  if (!latest || !(Date.now() - publishedAt < RECENT_MS)) return Response.json({ revalidated: false });
  if (latest !== refreshedFor) {
    refreshedFor = latest;
    revalidateTag(SANITY_TAG, { expire: 0 });
  }
  return Response.json({ revalidated: true, publishedAt: latest });
}
